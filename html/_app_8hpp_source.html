44constexpr uint64_t flowHashBytes(std::string_view text) noexcept {
45 uint64_t hash = kFlowFnvOffsetBasis;
46 for (const char c : text) {
47 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(c));
48 hash *= kFlowFnvPrime;
49 }
50 return (hash == 0ull) ? 1ull : hash;
51}

146inline Clay_Color Flow_Color(std::string_view hexRgba)
147{
148 if (hexRgba.size() != 9 || hexRgba[0] != '#') {
149 throw std::invalid_argument("Flow_Color expects #RRGGBBAA.");
150 }
151
152 const auto decodeHexNibble = [](char c) -> uint8_t {
153 if (c >= '0' && c <= '9') return static_cast<uint8_t>(c - '0');
154 if (c >= 'a' && c <= 'f') return static_cast<uint8_t>(10 + (c - 'a'));
155 if (c >= 'A' && c <= 'F') return static_cast<uint8_t>(10 + (c - 'A'));
156 throw std::invalid_argument("Flow_Color received invalid hex digit.");
157 };
158
159 const auto decodeHexByte = [&](std::size_t index) -> float {
160 const uint8_t hi = decodeHexNibble(hexRgba[index]);
161 const uint8_t lo = decodeHexNibble(hexRgba[index + 1]);
162 return static_cast<float>((hi << 4) | lo);
163 };
164
165 return Clay_Color{
166 decodeHexByte(1),
167 decodeHexByte(3),
168 decodeHexByte(5),
169 decodeHexByte(7),
170 };

122 explicit GlfwWindowBackend(const FlowUi::WindowConfig& config, InputQueue* inputQueue)
123 : input(inputQueue) {
124 glfwWindowHint(GLFW_CLIENT_API, GLFW_NO_API);
125 glfwWindowHint(GLFW_RESIZABLE, config.resizable ? GLFW_TRUE : GLFW_FALSE);
126 if (config.maximized) {
127 glfwWindowHint(GLFW_MAXIMIZED, GLFW_TRUE);
128 }
129#ifdef GLFW_SCALE_TO_MONITOR
130 glfwWindowHint(GLFW_SCALE_TO_MONITOR, config.highDPI ? GLFW_TRUE : GLFW_FALSE);
131#endif
132#ifdef GLFW_COCOA_RETINA_FRAMEBUFFER
133 glfwWindowHint(GLFW_COCOA_RETINA_FRAMEBUFFER, config.highDPI ? GLFW_TRUE : GLFW_FALSE);
134#endif
135
136 GLFWmonitor* monitor = nullptr;
137 int width = std::max(1, config.width);
138 int height = std::max(1, config.height);
139 if (config.fullscreen) {
140 monitor = glfwGetPrimaryMonitor();
141 if (!monitor) {
142 throw std::runtime_error("Failed to acquire primary monitor for fullscreen mode.");
143 }
144 const GLFWvidmode* mode = glfwGetVideoMode(monitor);
145 if (mode) {
146 width = mode->width;
147 height = mode->height;
148 }
149 }
150
151 window = glfwCreateWindow(width, height, config.title.c_str(), monitor, nullptr);
152 if (!window) {
153 throw std::runtime_error("Failed to create GLFW window.");
154 }
155
156 glfwSetWindowUserPointer(window, this);
157 installCallbacks();
158 setInputConfig(config.input);
159 setCursorType(FlowUi::CursorType::Arrow);
160 }

171 glfwPollEvents();

176 double mouseX = 0.0;

178 glfwGetCursorPos(window, &mouseX, &mouseY);

180
181 for (int mouseButton = 0; mouseButton < static_cast<int>(FrameInput::kMouseButtonCount); ++mouseButton) {
182 const int buttonState = glfwGetMouseButton(window, mouseButton);
183 input->pushMouseButton(mouseButton, buttonState == GLFW_PRESS);

314 GLFWcursor* acquireStandardCursor(FlowUi::CursorType cursorType) {
315 const std::size_t cursorIndex = static_cast<std::size_t>(cursorType);
316 if (cursorIndex >= standardCursors_.size()) {
317 return nullptr;
318 }
319
320 GLFWcursor*& cachedCursor = standardCursors_[cursorIndex];
321 if (cachedCursor) {
322 return cachedCursor;
323 }
324
325 cachedCursor = glfwCreateStandardCursor(toGlfwCursorShape(cursorType));
326 return cachedCursor;
327 }

329 void destroyStandardCursors() {
330 for (GLFWcursor*& cursor : standardCursors_) {

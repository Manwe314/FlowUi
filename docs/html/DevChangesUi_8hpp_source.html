36template <class Context>
37void register_focus_target(Context& context, DevChangesState& changes, Clay_ElementId identity,
38 bool enabled = true, FlowElementID text_input = {},
39 uint64_t* radio_selection = nullptr, uint64_t radio_value = 0,
40 Clay_ElementId viewport = {}) {
41 if (!enabled || !identity.id)
42 return;
43 const auto found = std::ranges::find(changes.focus_targets, identity.id,
44 [](const auto& target) { return target.identity.id; });
45 const DevChangesFocusTarget target{identity,
46 viewport.id ? viewport : changes.registration_viewport,
47 text_input, radio_selection, radio_value};
48 if (found == changes.focus_targets.end())
49 changes.focus_targets.emplace_back(target);
50 else
51 *found = target;
52 if (context.uiManager.getPreviousFramesInteraction().isPressed(identity)) {
53 context.uiManager.shortcuts().setFocusedElement(identity);
54 if (!text_input.value && context.uiManager.inputFields().hasPrimaryFieldFocus())
55 context.uiManager.inputFields().requestCaret(FlowElementID{},
56 CaretRequestKind::ClearAll);
57 }
58}

71template <class Context>
72void begin_focus_frame(Context& context, DevChangesState& changes) {
73 const auto& input = context.uiManager.getCurrentFrameInput();
74 const auto& previous = context.uiManager.getPreviousFrameInput();
75 const auto focused = context.uiManager.shortcuts().focusedElement().id;
76 const auto found = std::ranges::find(changes.focus_targets, focused,
77 [](const auto& target) { return target.identity.id; });
78 if (input.windowFocused && !changes.focus_targets.empty() && input.keyDown[GLFW_KEY_TAB] &&
79 !previous.keyDown[GLFW_KEY_TAB]) {
80 const bool reverse =
81 input.keyDown[GLFW_KEY_LEFT_SHIFT] || input.keyDown[GLFW_KEY_RIGHT_SHIFT];
82 const auto count = changes.focus_targets.size();
83 const auto position = found == changes.focus_targets.end()
84 ? (reverse ? 0 : count - 1)
85 : static_cast<std::size_t>(found - changes.focus_targets.begin());
86 const auto& target =
87 changes.focus_targets[(position + count + (reverse ? count - 1 : 1)) % count];
88 context.uiManager.inputFields().requestCaret(FlowElementID{}, CaretRequestKind::ClearAll);
89 context.uiManager.shortcuts().setFocusedElement(target.identity);
90 if (target.text_input.value)
91 context.uiManager.inputFields().requestCaret(target.text_input,
92 CaretRequestKind::SetPrimary);
93 if (target.viewport.id) {
94 const auto bounds = Clay_GetElementData(target.identity);
95 const auto viewport = Clay_GetElementData(target.viewport);
96 const auto scroll = Clay_GetScrollContainerData(target.viewport);
97 if (bounds.found && viewport.found && scroll.found && scroll.scrollPosition) {
98 if (bounds.boundingBox.y < viewport.boundingBox.y)
99 scroll.scrollPosition->y += viewport.boundingBox.y - bounds.boundingBox.y;
100 else if (bounds.boundingBox.y + bounds.boundingBox.height >
101 viewport.boundingBox.y + viewport.boundingBox.height)
102 scroll.scrollPosition->y -= bounds.boundingBox.y + bounds.boundingBox.height -
103 viewport.boundingBox.y -
104 viewport.boundingBox.height;
105 }
106 }
107 } else if (input.windowFocused && found != changes.focus_targets.end() &&
108 found->radio_selection && !context.uiManager.inputFields().hasPrimaryFieldFocus() &&
109 ((input.keyDown[GLFW_KEY_ENTER] && !previous.keyDown[GLFW_KEY_ENTER]) ||
110 (input.keyDown[GLFW_KEY_SPACE] && !previous.keyDown[GLFW_KEY_SPACE]))) {
111 *found->radio_selection = found->radio_value;
112 changes.model_dirty = true;
113 }
114 changes.focus_targets.clear();
115 if (changes.focus_targets.capacity() < 256)
116 changes.focus_targets.reserve(256);
117 changes.registration_viewport = {};
118}

189inline void activate(void* target, uint64_t action, uint64_t value) {
190 auto& state = *static_cast<DevChangesState*>(target);
191 if (action == Select || action == Toggle) {
192 const auto found = std::ranges::find(state.nodes, value, &DevChangesNode::identity);
193 if (found != state.nodes.end()) {
194 state.focused_identity = value;
195 state.keyboard_active = true;
196 if (action == Toggle && !found->leaf)
197 found->expanded = !found->expanded;
198 if (action == Select) {
199 changes_select_key(state, found->key);
200 }
201 state.model_dirty = true;
202 }
203 }
204 if (action == Refresh)
205 state.pending = DevChangesCommand::Refresh;
206 if (action == Bake)
207 state.pending = DevChangesCommand::Bake;
208 if (action == Include)
209 queue_changes_decision(state, tooling::DevChangesDecision::IncludeProposal);
210 if (action == Omit)
211 queue_changes_decision(state, tooling::DevChangesDecision::OmitProposal);
212 if (action == Keep)
213 queue_changes_decision(state, tooling::DevChangesDecision::RetainDisk);
214 if (action == Remove)
215 queue_changes_decision(state, tooling::DevChangesDecision::Remove);
216 if (action == Undo)
217 state.pending = DevChangesCommand::Undo;
218 if (action == Redo)
219 state.pending = DevChangesCommand::Redo;
220 if (action == Edit)
221 state.pending = DevChangesCommand::Edit;
222 if (action == Back)
223 state.pending = DevChangesCommand::Back;
224 if (action == ApplyProposal)
225 state.pending = DevChangesCommand::ApplyProposal;
226 if (action == ApplySaved)
227 state.pending = DevChangesCommand::ApplySaved;
228 if (action == DiscardLive)
229 state.pending = DevChangesCommand::DiscardLive;
230 if (action == ResetAuthored)
231 state.pending = DevChangesCommand::ResetAuthored;
232 if (action == UndoLive)
233 state.pending = DevChangesCommand::UndoLive;
234 if (action == RedoLive)
235 state.pending = DevChangesCommand::RedoLive;
236 if (action == StagePolicy) {
237 if (value == 1 && !state.policy_growth_valid) {
238 state.message = "Complete a valid growth factor draft before staging.";
239 state.message_is_error = true;
240 return;
241 }
242 state.pending_policy = value;
243 state.pending = DevChangesCommand::StagePolicy;
244 }
245 if (action == PreviousPage && state.workbench_page) {
246 --state.workbench_page;
247 state.workbench_page_changed = true;
248 }
249 if (action == NextPage) {
250 ++state.workbench_page;
251 state.workbench_page_changed = true;
252 }
253 if (action == Overview) {
254 state.selected_key.clear();
255 state.queue_scope_key.clear();
256 state.selected_outside_filter = false;
257 state.workbench_page = 0;
258 state.workbench_page_changed = true;
259 state.group_decisions_open = false;
260 }
261 if (action == Section && value < state.inspector_expanded.size())
262 state.inspector_expanded[value] = !state.inspector_expanded[value];
263 if (action == Review) {
264 state.view = 1;
265 state.group_decisions_open = false;
266 }
267 if (action == CloseReview)
268 state.view = 0;
269 if (action == Issues) {
270 state.status = 3;
271 state.domain = 0;
272 state.search.clear();
273 state.model_dirty = true;
274 }
275 if (action == Acknowledge) {
276 state.acknowledge_incomplete = !state.acknowledge_incomplete;
277 state.pending = DevChangesCommand::Refresh;
278 }
279 if (action == ClearSelection) {
280 state.selected_key.clear();
281 state.selected_outside_filter = false;
282 }
283 if (action == RevealSelection) {
284 state.search.clear();
285 state.status = state.domain = 0;
286 std::string ancestor_key = state.selected_key;
287 while (!ancestor_key.empty()) {
288 const auto ancestor =
289 std::ranges::find(state.nodes, ancestor_key, &DevChangesNode::key);
290 if (ancestor == state.nodes.end())
291 break;
292 if (!ancestor->leaf)
293 ancestor->expanded = true;
294 ancestor_key = ancestor->parent;
295 }
296 state.reveal_selection_pending = true;
297 state.model_dirty = true;
298 }
299 if (action == NextIssue && !state.review.rows.empty()) {
300 const auto selected = std::ranges::find(state.review.rows, state.selected_key,
301 &tooling::DevChangesReviewRow::key);
302 const auto start = selected == state.review.rows.end()
303 ? 0
304 : static_cast<std::size_t>(selected - state.review.rows.begin()) + 1;
305 for (std::size_t offset = 0; offset < state.review.rows.size(); ++offset) {
306 const auto& candidate = state.review.rows[(start + offset) % state.review.rows.size()];
307 if (candidate.blocking) {
308 changes_select_key(state, candidate.key);
309 state.view = 0;
310 break;
311 }
312 }
313 }
314 if (action == GroupDecisions)
315 state.group_decisions_open = !state.group_decisions_open;
316 if (action == CloseGroupDecisions)
317 state.group_decisions_open = false;
318 if (action == OpenPolicy) {
319 state.view = 2;
320 state.group_decisions_open = false;
321 }
322 if (action == ClosePolicy)
323 state.view = 0;
324 if (action == ToggleRawManifest)
325 state.raw_manifest_open = !state.raw_manifest_open;
326 if (action == ToggleOutputTechnical)
327 state.output_technical_open = !state.output_technical_open;
328 if (action == DismissMessage) {
329 state.message.clear();
330 state.message_is_error = false;
331 }
332}

349template <class Context>
350[[nodiscard]] std::string ellipsize(Context& context, std::string_view label, float width,
351 uint16_t size = 12) {
352 if (width <= 0)
353 return {};
354 const auto* face = text_face(context);
355 if (!face) {
356 const auto limit = static_cast<std::size_t>(width / (size * 1.1f));
357 return label.size() <= limit ? std::string(label)
358 : limit > 3 ? std::string(label.substr(0, limit - 3)) + "..."
359 : std::string{};
360 }
361 const ::FlowUi::detail::text::CurrentTextShaper shaper{};
362 auto layout = shaper.shape({.text = label, .pointsToPixelsScale = 1, .fontSize = size}, *face);
363 Clay_TextElementConfig config{};
364 config.fontSize = size;
365 const float height =
366 context.uiManager.inputContentElement(config).layout.sizing.height.size.minMax.min;
367 const float scale = layout.lineHeight > 0 ? height / layout.lineHeight : 1;
368 if (layout.measuredWidth * scale <= width - 2)
369 return std::string(label);
370 const auto suffix =
371 shaper.shape({.text = "...", .pointsToPixelsScale = 1, .fontSize = size}, *face);
372 const float available = width - suffix.measuredWidth * scale - 3;
373 std::size_t prefix = 0;
374 for (const auto& cluster : layout.clusters) {
375 if (cluster.xEnd * scale > available)
376 break;
377 prefix = cluster.endByte;
378 }
379 return prefix ? std::string(label.substr(0, prefix)) + "..." : std::string{};
380}

383template <class Context>
384void text(Context& context, std::string_view label,
385 Clay_Color color = interface_theme::kTextSecondary, uint16_t size = 12) {
386 std::string display;
387 display.reserve(label.size() + label.size() / 24);
388 const auto parent = Clay_GetElementData(Clay_ElementId{.id = Clay_GetOpenElementId()});
389 const float available = std::max(
390 1.0f, parent.found && parent.boundingBox.width > 30 ? parent.boundingBox.width - 30 : 180);
391 Clay_TextElementConfig config{};
392 config.fontSize = size;
393 const float resolved_height =
394 context.uiManager.inputContentElement(config).layout.sizing.height.size.minMax.min;
395 const auto* face = text_face(context);
396 const ::FlowUi::detail::text::CurrentTextShaper shaper{};
397 float scale = 1;
398 if (face) {
399 const auto metrics =
400 shaper.shape({.text = "M", .pointsToPixelsScale = 1, .fontSize = size}, *face);
401 if (metrics.lineHeight > 0)
402 scale = resolved_height / metrics.lineHeight;
403 }
404 const auto whitespace = [](char character) {
405 return character == ' ' || character == '\n' || character == '\t' || character == '\r';
406 };
407 std::size_t offset = 0;
408 while (offset < label.size()) {
409 if (whitespace(label[offset])) {
410 display += label[offset++];
411 continue;
412 }
413 const auto word_end = label.find_first_of(" \n\t\r", offset);
414 const auto end = word_end == std::string_view::npos ? label.size() : word_end;
415 const auto word = label.substr(offset, end - offset);
416 auto layout =
417 face ? shaper.shape({.text = word, .pointsToPixelsScale = 1, .fontSize = size}, *face)
418 : ::FlowUi::detail::text::TextLayoutResult{};
419 if (face && layout.success && !layout.clusters.empty()) {
420 if (layout.measuredWidth * scale <= available)
421 display += word;
422 else {
423 std::size_t first_cluster = 0;
424 while (first_cluster < layout.clusters.size()) {
425 std::size_t after_cluster = first_cluster + 1;
426 const float start_x = layout.clusters[first_cluster].xStart;
427 while (after_cluster < layout.clusters.size() &&
428 (layout.clusters[after_cluster].xEnd - start_x) * scale <= available - 2)
429 ++after_cluster;
430 const auto first_byte = layout.clusters[first_cluster].startByte;
431 auto segment = word.substr(
432 first_byte, layout.clusters[after_cluster - 1].endByte - first_byte);
433 while (after_cluster > first_cluster + 1) {
434 const auto measured = shaper.shape(
435 {.text = segment, .pointsToPixelsScale = 1, .fontSize = size}, *face);
436 if (measured.measuredWidth * scale <= available)
437 break;
438 --after_cluster;
439 segment = word.substr(
440 first_byte, layout.clusters[after_cluster - 1].endByte - first_byte);
441 }
442 display += segment;
443 first_cluster = after_cluster;
444 if (first_cluster < layout.clusters.size())
445 display += '\n';
446 }
447 }
448 } else {
449 const auto byte_limit = static_cast<std::size_t>(
450 std::max(1.0f, available / std::max(1.0f, resolved_height * .5f)));
451 std::size_t word_offset = 0, line_bytes = 0;
452 while (word_offset < word.size()) {
453 auto character_end = word_offset + 1;
454 while (character_end < word.size() &&
455 (static_cast<unsigned char>(word[character_end]) & 0xc0u) == 0x80u)
456 ++character_end;
457 const auto character_bytes = character_end - word_offset;
458 if (line_bytes && line_bytes + character_bytes > byte_limit) {
459 display += '\n';
460 line_bytes = 0;
461 }
462 display += word.substr(word_offset, character_bytes);
463 line_bytes += character_bytes;
464 word_offset = character_end;
465 }
466 }
467 offset = end;
468 }
469 diagnostics_ui::text(context, display, color, size);
470}

595template <class Context>
596[[nodiscard]] Clay_ElementDeclaration viewport(Context& context, LocalElementName identity,
597 float& retained_scroll,
598 ViewportState* restoration = nullptr) noexcept {
599 auto result = column(0);
600 result.layout.sizing.height = CLAY_SIZING_GROW(0);
601 const auto scroll = Clay_GetScrollContainerData(context.clayID(identity));
602 if (restoration && !scroll.found) {
603 restoration->requested_offset = retained_scroll;
604 restoration->settling_frames = retained_scroll < 0 ? 4 : 0;
605 }
606 if (restoration && restoration->settling_frames) {
607 const auto& input = context.uiManager.getCurrentFrameInput();
608 const float lower_bound = std::min(0.0f, scroll.scrollContainerDimensions.height -
609 scroll.contentDimensions.height);
610 const bool moved_without_clamping =
611 scroll.found && scroll.scrollPosition &&
612 std::abs(scroll.scrollPosition->y - restoration->requested_offset) > 0.5f &&
613 std::abs(scroll.scrollPosition->y - lower_bound) > 0.5f;
614 const bool navigating = input.mouseDown[0] || input.scrollY != 0 || input.scrollX != 0 ||
615 input.keyDown[GLFW_KEY_TAB] || input.keyDown[GLFW_KEY_UP] ||
616 input.keyDown[GLFW_KEY_DOWN] || input.keyDown[GLFW_KEY_PAGE_UP] ||
617 input.keyDown[GLFW_KEY_PAGE_DOWN] || input.keyDown[GLFW_KEY_HOME] ||
618 input.keyDown[GLFW_KEY_END] || input.keyDown[GLFW_KEY_ENTER] ||
619 input.keyDown[GLFW_KEY_SPACE];
620 if (navigating || moved_without_clamping)
621 restoration->settling_frames = 0;
622 }
623 if (restoration && restoration->settling_frames) {
624 retained_scroll = restoration->requested_offset;
625 if (scroll.found && scroll.scrollPosition)
626 scroll.scrollPosition->y = retained_scroll;

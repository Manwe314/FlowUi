107template <auto Member>
108struct DevMemberOps {
109 using Traits = MemberPointerTraits<decltype(Member)>;
110 using Owner = typename Traits::Owner;
111 using Value = typename Traits::Value;
112
113 static const void* constAddress(const void* owner) noexcept {
114 if (owner == nullptr) return nullptr;
115 return std::addressof(static_cast<const Owner*>(owner)->*Member);
116 }
117
118 static void* mutableAddress(void* owner) noexcept {
119 if (owner == nullptr) return nullptr;
120 return std::addressof(static_cast<Owner*>(owner)->*Member);
121 }
122
123 static DevValueOperationStatus copyConstructMember(
124 const void* owner,
125 void* destination) noexcept {
126 if (owner == nullptr) return DevValueOperationStatus::NullSource;
127 if (destination == nullptr) return DevValueOperationStatus::NullDestination;
128 if constexpr (!std::is_copy_constructible_v<Value>) {
129 return DevValueOperationStatus::Unsupported;
130 } else {
131 try {
132 std::construct_at(
133 static_cast<Value*>(destination),
134 static_cast<const Owner*>(owner)->*Member);
135 return DevValueOperationStatus::Success;
136 } catch (...) {
137 return DevValueOperationStatus::Failed;
138 }
139 }
140 }
141
142 static DevValueOperationStatus assignMemberFromCopy(
143 void* owner,
144 const void* source) noexcept {
145 if (owner == nullptr) return DevValueOperationStatus::NullDestination;
146 if (source == nullptr) return DevValueOperationStatus::NullSource;
147 if constexpr (std::is_const_v<Value> || !std::is_copy_assignable_v<Value>) {
148 return DevValueOperationStatus::Unsupported;
149 } else {
150 try {
151 static_cast<Owner*>(owner)->*Member = *static_cast<const Value*>(source);
152 return DevValueOperationStatus::Success;
153 } catch (...) {
154 return DevValueOperationStatus::Failed;
155 }
156 }
157 }
158
159 static constexpr DevFieldOps operations{
160 .constAddress = &constAddress,
161 .mutableAddress = std::is_const_v<Value> ? nullptr : &mutableAddress,
162 .copyConstructMember = &copyConstructMember,
163 .assignMemberFromCopy = &assignMemberFromCopy,
164 };
165};

349template <typename T>
350concept HasAdlDevEnumSchema = requires {
351 flowUiDevEnumSchema(DevSchemaTag<T>{});
352};
353
354template <typename T>
355consteval auto declaredSchema() {
356 static_assert(!(HasAdlDevSchema<T> && HasIntrusiveDevSchema<T>),
357 "FlowUi DEV type declares both ADL flowUiDevSchema and T::devSchema().");
358 if constexpr (HasAdlDevSchema<T>) {
359 return flowUiDevSchema(DevSchemaTag<T>{});
360 } else if constexpr (HasIntrusiveDevSchema<T>) {
361 return T::devSchema();
362 } else {
363 return DevTypeAdapter<T>::schema();
364 }
365}
366
367template <typename T>
368inline constexpr bool hasDeclaredSchema =
369 DevTypeAdapter<T>::enabled || HasAdlDevSchema<T> || HasIntrusiveDevSchema<T>;
370
371template <typename T>
372inline constexpr bool hasDeclaredEnumSchema =
373 DevEnumAdapter<T>::enabled || HasAdlDevEnumSchema<T>;
374
375template <typename T>
376consteval auto declaredEnumSchema() {
377 static_assert(std::is_enum_v<T>);
378 static_assert(!(HasAdlDevEnumSchema<T> && DevEnumAdapter<T>::enabled),
379 "FlowUi DEV enum declares both an ADL schema and a built-in adapter.");
380 if constexpr (HasAdlDevEnumSchema<T>) {
381 return flowUiDevEnumSchema(DevSchemaTag<T>{});
382 } else {
383 return DevEnumAdapter<T>::schema();
384 }
385}

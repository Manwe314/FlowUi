90template <auto Member>
91struct DevMemberOps {
92 using Traits = MemberPointerTraits<decltype(Member)>;
93 using Owner = typename Traits::Owner;
94 using Value = typename Traits::Value;
95
96 static const void* constAddress(const void* owner) noexcept {
97 if (owner == nullptr) return nullptr;
98 return std::addressof(static_cast<const Owner*>(owner)->*Member);
99 }
100
101 static void* mutableAddress(void* owner) noexcept {
102 if (owner == nullptr) return nullptr;
103 return std::addressof(static_cast<Owner*>(owner)->*Member);
104 }
105
106 static DevValueOperationStatus copyConstructMember(
107 const void* owner,
108 void* destination) noexcept {
109 if (owner == nullptr) return DevValueOperationStatus::NullSource;
110 if (destination == nullptr) return DevValueOperationStatus::NullDestination;
111 if constexpr (!std::is_copy_constructible_v<Value>) {
112 return DevValueOperationStatus::Unsupported;
113 } else {
114 try {
115 std::construct_at(
116 static_cast<Value*>(destination),
117 static_cast<const Owner*>(owner)->*Member);
118 return DevValueOperationStatus::Success;
119 } catch (...) {
120 return DevValueOperationStatus::Failed;
121 }
122 }
123 }
124
125 static DevValueOperationStatus assignMemberFromCopy(
126 void* owner,
127 const void* source) noexcept {
128 if (owner == nullptr) return DevValueOperationStatus::NullDestination;
129 if (source == nullptr) return DevValueOperationStatus::NullSource;
130 if constexpr (std::is_const_v<Value> || !std::is_copy_assignable_v<Value>) {
131 return DevValueOperationStatus::Unsupported;
132 } else {
133 try {
134 static_cast<Owner*>(owner)->*Member = *static_cast<const Value*>(source);
135 return DevValueOperationStatus::Success;
136 } catch (...) {
137 return DevValueOperationStatus::Failed;
138 }
139 }
140 }
141
142 static constexpr DevFieldOps operations{
143 .constAddress = &constAddress,
144 .mutableAddress = std::is_const_v<Value> ? nullptr : &mutableAddress,
145 .copyConstructMember = &copyConstructMember,
146 .assignMemberFromCopy = &assignMemberFromCopy,
147 };

301template <typename T>
302consteval auto declaredSchema() {
303 static_assert(!(HasAdlDevSchema<T> && HasIntrusiveDevSchema<T>),
304 "FlowUi DEV type declares both ADL flowUiDevSchema and T::devSchema().");
305 if constexpr (HasAdlDevSchema<T>) {
306 return flowUiDevSchema(DevSchemaTag<T>{});
307 } else if constexpr (HasIntrusiveDevSchema<T>) {
308 return T::devSchema();
309 } else {
310 return DevTypeAdapter<T>::schema();
311 }
312}

314template <typename T>
315inline constexpr bool hasDeclaredSchema =
316 DevTypeAdapter<T>::enabled || HasAdlDevSchema<T> || HasIntrusiveDevSchema<T>;
317
318template <typename T>
319inline constexpr bool hasDeclaredEnumSchema =
320 DevEnumAdapter<T>::enabled || HasAdlDevEnumSchema<T>;
321
322template <typename T>
323consteval auto declaredEnumSchema() {
324 static_assert(std::is_enum_v<T>);
325 static_assert(!(HasAdlDevEnumSchema<T> && DevEnumAdapter<T>::enabled),
326 "FlowUi DEV enum declares both an ADL schema and a built-in adapter.");
327 if constexpr (HasAdlDevEnumSchema<T>) {
328 return flowUiDevEnumSchema(DevSchemaTag<T>{});
329 } else {
330 return DevEnumAdapter<T>::schema();
331 }

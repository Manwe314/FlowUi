221inline constexpr uint64_t kHashPrime = 1099511628211ull;

223[[nodiscard]] consteval uint64_t hashBytes(std::string_view value, uint64_t hash = kHashOffset) {
224 for (const char character : value) {
225 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(character));
226 hash *= kHashPrime;
227 }
228 return hash;
229}
230
231[[nodiscard]] consteval uint64_t hashInteger(uint64_t value, uint64_t hash) {
232 for (uint32_t byteIndex = 0u; byteIndex < sizeof(value); ++byteIndex) {
233 hash ^= (value >> (byteIndex * 8u)) & 0xFFu;
234 hash *= kHashPrime;
235 }
236 return hash;
237}

239[[nodiscard]] consteval TimingZoneTypeId makeZoneTypeId(
240 TimingCategory category,
241 std::string_view name,
242 std::string_view file,
243 uint32_t line) {
244 uint64_t hash = hashBytes("FlowUi.DevTiming.Zone");
245 hash = hashInteger(static_cast<uint64_t>(category), hash);
246 hash = hashBytes(name, hash);
247 hash = hashBytes(file, hash);
248 hash = hashInteger(line, hash);
249 return hash == 0u ? 1u : hash;
250}

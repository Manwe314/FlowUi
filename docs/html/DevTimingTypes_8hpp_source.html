362inline constexpr uint64_t kHashOffset = 14695981039346656037ull;
363inline constexpr uint64_t kHashPrime = 1099511628211ull;
364
365[[nodiscard]] consteval uint64_t hashBytes(std::string_view value, uint64_t hash = kHashOffset) {
366 for (const char character : value) {
367 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(character));
368 hash *= kHashPrime;
369 }
370 return hash;
371}
372
373[[nodiscard]] consteval uint64_t hashInteger(uint64_t value, uint64_t hash) {
374 for (uint32_t byteIndex = 0u; byteIndex < sizeof(value); ++byteIndex) {
375 hash ^= (value >> (byteIndex * 8u)) & 0xFFu;
376 hash *= kHashPrime;
377 }
378 return hash;
379}

381[[nodiscard]] consteval TimingZoneTypeId makeZoneTypeId(
382 TimingCategory category,
383 std::string_view name,
384 std::string_view file,
385 uint32_t line) {
386 uint64_t hash = hashBytes("FlowUi.DevTiming.Zone");
387 hash = hashInteger(static_cast<uint64_t>(category), hash);
388 hash = hashBytes(name, hash);
389 hash = hashBytes(file, hash);
390 hash = hashInteger(line, hash);
391 return hash == 0u ? 1u : hash;
392}

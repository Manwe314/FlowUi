661 template <OutputMode Mode>
662 void invoke(ElementDrawOptions options) {
663 FlowScopeGuard flowScope{uiManager_, elementId_};
664
665#if FLOW_UI_DEV_MODE
666 if (uiManager_.isDevPassivePreview())
667 options = options | ElementDrawOptions::SkipEventCallbacks |
668 ElementDrawOptions::SkipLogicCallback;
669 if constexpr (!detail::element::isDevInternal<ElementType>())
670 if (auto* registry = uiManager_.devSchemaRegistry())
671 registry->template registerElementPreview<ElementType>();
672 // Route header-only builder capture through the internal runtime bridge.
673 detail::claimFlowRootForDev(
674 uiManager_,
675 elementId_,
676 ElementType::definitionId,
677 sourceLocation_.file_name(),
678 static_cast<uint32_t>(sourceLocation_.line()),
679 static_cast<uint32_t>(sourceLocation_.column()),
680 sourceLocation_.function_name(),
681 automaticIdentity_);
682#endif
683
684#if FLOW_UI_DEV_MODE && FLOWUI_DEV_TIMING_LEVEL >= 1
685 devSystems::ElementTimingZone invocationTiming(
686 uiManager_.devTimingRecorder(), ElementType::definitionId, elementId_);
687#endif
688 auto invocation = [&]() {
689#if FLOW_UI_DEV_MODE
690 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(
691 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
692 devSystems::TimingZoneRole::Work, "flowui.element.registration_state");
693#endif
694 return detail::element::ElementInvocation<ElementType>::begin(
695 elementManager_, uiManager_, window_, elementId_);
696 }();
697
698#if FLOW_UI_DEV_MODE
699 const bool isDevInternal =
700 captureAsDevInternal_ || uiManager_.isDevInternalScope();
701 DevInternalScopeGuard devInternalGuard{uiManager_, isDevInternal};
702 auto* overrides = uiManager_.devOverrideEngine();
703 auto authored_parameters = !isDevInternal && overrides
704 ? overrides->captureAuthoredParameters(ElementType::definitionId, std::addressof(params_))
705 : devSystems::tooling::DevAuthoredElementCapture{};
706#endif
707
708#if FLOW_UI_HAS_BAKED_CHANGES
709#if FLOW_UI_DEV_MODE
710 if (!isDevInternal)
711#endif
712 baked::applyBakedElementDefaults<ElementType>(
713 detail::element::toInstanceKey(elementId_), params_);
714#endif
715
716#if FLOW_UI_DEV_MODE
717 if (!isDevInternal && overrides) {
718 overrides->applyElement(ElementType::definitionId, window_,
719 detail::element::toInstanceKey(elementId_), std::addressof(params_),
720 uiManager_.devTimingRecorder());
721 }
722#endif
723
724 Clay_ElementId rootElementId{};
725 if constexpr (Mode == OutputMode::Construct) {
726 rootElementId = uiManager_.toClayEID(elementId_);
727 }
728
729 ConstructedDepthGuard constructedDepth{uiManager_};
730
731 {
732#if FLOW_UI_DEV_MODE
733 FLOWUI_DEV_TIMING_ZONE_BALANCED_IF(
734 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
735 devSystems::TimingZoneRole::Work, "flowui.element.interaction_hooks");
736#endif
737 invokeInteractionHooks(invocation, elementId_, options);
738 }
739
740 if constexpr (Mode == OutputMode::Draw) {
741 if (elementDrawOptionsHas(options, ElementDrawOptions::SkipBuildCallback)) {
742#if FLOW_UI_DEV_MODE && FLOWUI_DEV_TIMING_LEVEL >= 1
743 invocationTiming.end();
744#endif
745 return;
746 }
747 }
748 BuildContext buildContext{invocation, params_};
749#if FLOW_UI_DEV_MODE
750 const auto treeToken = detail::dev_tooling_bridge::beginCapturedFlowElement(
751 uiManager_, ElementType::definitionId, detail::element::debugName<ElementType>(),
752 detail::typeToken<ElementType>(), elementId_, isDevInternal,
753 Mode == OutputMode::Construct, sourceLocation_.file_name(),
754 static_cast<uint32_t>(sourceLocation_.line()),
755 static_cast<uint32_t>(sourceLocation_.column()), sourceLocation_.function_name());
756 DevCaptureScope devCapture{&uiManager_, treeToken};
757#endif
758 if constexpr (Mode == OutputMode::Draw) {
759 {
760#if FLOW_UI_DEV_MODE
761 FLOWUI_DEV_TIMING_ZONE_BALANCED_IF(
762 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
763 devSystems::TimingZoneRole::Work, "flowui.element.build_callback");
764#endif
765 ElementType::buildElement(buildContext);
766 }
767#if FLOW_UI_DEV_MODE
768 captureEffectiveParameters(treeToken, invocation, std::move(authored_parameters));
769#endif
770 } else {
771 Clay_ElementDeclaration declaration{};
772 {
773#if FLOW_UI_DEV_MODE
774 FLOWUI_DEV_TIMING_ZONE_BALANCED_IF(
775 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
776 devSystems::TimingZoneRole::Work, "flowui.element.construct_callback");
777#endif
778 declaration = ElementType::constructElement(buildContext);
779 }
780#if FLOW_UI_DEV_MODE
781 captureEffectiveParameters(treeToken, invocation, std::move(authored_parameters));
782#endif
783 if (uiManager_.constructedElementDepth() != constructedDepth.baseline) {
784 detail::terminateForFatalError(makeError(ErrorCode::InternalInvariantBroken, ErrorSite::ElementInvoke));
785 }
786#if FLOW_UI_DEV_MODE && FLOWUI_DEV_TIMING_LEVEL >= 1
787 invocationTiming.end();
788#endif
789 uiManager_.retainConstructedElement(
790 rootElementId, elementId_, flowScope.priorDepth
791#if FLOW_UI_DEV_MODE
792 , treeToken
793 , isDevInternal
794#endif
795#if FLOW_UI_DEV_MODE && FLOWUI_DEV_TIMING_LEVEL >= 2
796 , ElementType::definitionId
797#endif
798 );
799 Clay__OpenElementWithId(rootElementId);
800 Clay__ConfigureOpenElement(declaration);
801 constructedDepth.release();
802 flowScope.release();
803#if FLOW_UI_DEV_MODE
804 devCapture.leaveOpen();
805#endif
806 }
807#if FLOW_UI_DEV_MODE && FLOWUI_DEV_TIMING_LEVEL >= 1
808 if constexpr (Mode == OutputMode::Draw) invocationTiming.end();
809#endif
810 }

832 void invokeInteractionHooks(
833 detail::element::ElementInvocation<ElementType>& invocation,
834 FlowElementID elementId,
835 ElementDrawOptions options) {
836 if constexpr (!hasInteractionHooks) {
837 (void)invocation;
838 (void)elementId;
839 (void)options;
840 return;
841 } else {
842 bool invokeEvents = false;
843 bool invokeLogic = false;
844 if constexpr (hasEventHooks) {
845 invokeEvents = !elementDrawOptionsHas(
846 options, ElementDrawOptions::SkipEventCallbacks);
847 }
848 if constexpr (detail::element::HasRunLogicHook<ElementType>) {
849 invokeLogic = !elementDrawOptionsHas(
850 options, ElementDrawOptions::SkipLogicCallback);
851 }
852 if (!invokeEvents && !invokeLogic) return;
853
854 const InteractionSnapshot& previousInteraction =
855 uiManager_.getPreviousFramesInteraction();
856 InteractionContext context{
857 invocation, params_, previousInteraction};
858
859 if (invokeEvents) {
860 if constexpr (detail::element::HasOnHoveredHook<ElementType>) {
861 if (previousInteraction.isHovered(elementId)) {
862 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(
863 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
864 devSystems::TimingZoneRole::Work, "flowui.element.on_hovered");
865 context.setActionInvocationSourceKind(
866 ActionInvocationSourceKind::ElementHovered);
867 ElementType::onHovered(context);
868 }
869 }
870 if constexpr (detail::element::HasOnPressedHook<ElementType>) {
871 if (previousInteraction.isPressed(elementId)) {
872 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(
873 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
874 devSystems::TimingZoneRole::Work, "flowui.element.on_pressed");
875 context.setActionInvocationSourceKind(
876 ActionInvocationSourceKind::ElementPressed);
877 ElementType::onPressed(context);
878 }
879 }
880 if constexpr (detail::element::HasOnHeldHook<ElementType>) {
881 if (previousInteraction.isHeld(elementId)) {
882 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(
883 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
884 devSystems::TimingZoneRole::Work, "flowui.element.on_held");
885 context.setActionInvocationSourceKind(
886 ActionInvocationSourceKind::ElementHeld);
887 ElementType::onHeld(context);
888 }
889 }
890 if constexpr (detail::element::HasOnReleasedHook<ElementType>) {
891 if (previousInteraction.isReleased(elementId)) {
892 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(
893 uiManager_.devTimingRecorder(), devSystems::TimingCategory::Element,
894 devSystems::TimingZoneRole::Work, "flowui.element.on_released");
895 context.setActionInvocationSourceKind(
896 ActionInvocationSourceKind::ElementReleased);
897 ElementType::onReleased(context);
898 }
899 }
900 }
901
902 if constexpr (detail::element::HasRunLogicHook<ElementType>) {
903 if (invokeLogic) {
904 FLOWUI_DEV_TIMING_ZONE_DEEP_IF(

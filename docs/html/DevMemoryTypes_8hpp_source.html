417[[nodiscard]] consteval uint64_t hashBytes(std::string_view value, uint64_t hash = kHashOffset) {
418 for (const char character : value) {
419 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(character));
420 hash *= kHashPrime;
421 }
422 return hash;
423}

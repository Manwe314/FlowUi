267inline constexpr uint64_t kHashPrime = 1099511628211ull;

269[[nodiscard]] consteval uint64_t hashBytes(std::string_view value, uint64_t hash = kHashOffset) {
270 for (const char character : value) {
271 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(character));
272 hash *= kHashPrime;
273 }
274 return hash;
275}

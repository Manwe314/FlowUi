132inline constexpr uint64_t kHashOffset = 14695981039346656037ull;
133inline constexpr uint64_t kHashPrime = 1099511628211ull;
134
135[[nodiscard]] constexpr uint64_t hashBytes(
136 std::string_view value,
137 uint64_t hash = kHashOffset) noexcept {
138 for (const char character : value) {
139 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(character));
140 hash *= kHashPrime;
141 }
142 return hash;
143}
144
145[[nodiscard]] constexpr uint64_t hashInteger(uint64_t value, uint64_t hash) noexcept {
146 for (uint32_t byte = 0u; byte < sizeof(value); ++byte) {
147 hash ^= (value >> (byte * 8u)) & 0xffu;
148 hash *= kHashPrime;
149 }
150 return hash;
151}

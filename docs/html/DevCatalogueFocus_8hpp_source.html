63 !previous.keyDown[GLFW_KEY_ESCAPE] &&

72 state.dirty = true;

74 }

76 !state.focus_targets.empty() && !context.uiManager.popups().has_open_popups()) {
77 const auto focused = context.uiManager.shortcuts().focusedElement().id;
78 const auto found = std::ranges::find(state.focus_targets, focused,
79 [](const auto& target) { return target.identity.id; });
80 const bool reverse =
81 current.keyDown[GLFW_KEY_LEFT_SHIFT] || current.keyDown[GLFW_KEY_RIGHT_SHIFT];
82 const auto count = state.focus_targets.size();
83 const auto position = found == state.focus_targets.end()
84 ? (reverse ? 0 : count - 1)
85 : static_cast<size_t>(found - state.focus_targets.begin());
86 const auto& target = state.focus_targets[(position + (reverse ? count - 1 : 1)) % count];

88 context.uiManager.shortcuts().setFocusedElement(target.identity);

95 const auto scroll = Clay_GetScrollContainerData(target.viewport);
96 if (bounds.found && viewport.found && scroll.found && scroll.scrollPosition) {
97 if (bounds.boundingBox.y < viewport.boundingBox.y)
98 scroll.scrollPosition->y += viewport.boundingBox.y - bounds.boundingBox.y;
99 else if (bounds.boundingBox.y + bounds.boundingBox.height >
100 viewport.boundingBox.y + viewport.boundingBox.height)

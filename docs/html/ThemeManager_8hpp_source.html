90 [[nodiscard]] Status registerDevPreview(std::string_view debug_name,
91 const std::function<void(App&, UiManager&)>& draw,
92 bool uses_local_theme = false) noexcept {
93 if (debug_name.empty() || !draw)
94 return unexpectedError(
95 makeError(ErrorCode::ThemeMutationFailed, ErrorSite::ThemeRegisterVariant));

97 for (auto& preview : dev_previews_)

105 DevThemePreview{draw, std::string(debug_name), uses_local_theme},

107 return {};

109 return unexpectedError(detail::currentExceptionError(ErrorSite::ThemeRegisterVariant));

284 template <typename T>
285 [[nodiscard]] Status registerTheme(std::string_view variantName, const T& themeData,
286 bool makeActive = true) {
287 if (!controller_ || !storage_) {
288 return unexpectedError(makeError(ErrorCode::ObjectNotInitialized, ErrorSite::ThemeRegisterVariant));
289 }
290 const auto writable = detail::catchResult<Status>(ErrorSite::ThemeRegisterVariant, [&] {
291 storage_->noteManagerMutation(InvalidWindowId);
292 return Status{};
293 });
294 if (!writable) return writable;
295#if FLOW_UI_DEV_MODE
296 if (devSchemaRegistry_) devSchemaRegistry_->ensureTheme<T>();
297#endif
298 T ownedThemeData = themeData;
299#if FLOW_UI_DEV_MODE
300 const T authored = themeData;
301#endif
302#if FLOW_UI_HAS_BAKED_CHANGES
303 baked::applyBakedThemeDefaults<T>(variantName, ownedThemeData);
304#endif
305 const auto variantId = controller_->internString(variantName);
306 controller_->registerThemeVariant<T>(variantId, std::move(ownedThemeData), makeActive
307#if FLOW_UI_DEV_MODE
308 ,
309 &authored
310#endif
311 );
312 return {};
313 }

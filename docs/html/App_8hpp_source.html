45constexpr uint64_t flowHashBytes(std::string_view text) noexcept {
46 uint64_t hash = kFlowFnvOffsetBasis;
47 for (const char c : text) {
48 hash ^= static_cast<uint64_t>(static_cast<unsigned char>(c));
49 hash *= kFlowFnvPrime;
50 }
51 return (hash == 0ull) ? 1ull : hash;
52}

147inline Clay_Color Flow_Color(std::string_view hexRgba)
148{
149 if (hexRgba.size() != 9 || hexRgba[0] != '#') {
150 throw std::invalid_argument("Flow_Color expects #RRGGBBAA.");
151 }
152
153 const auto decodeHexNibble = [](char c) -> uint8_t {
154 if (c >= '0' && c <= '9') return static_cast<uint8_t>(c - '0');
155 if (c >= 'a' && c <= 'f') return static_cast<uint8_t>(10 + (c - 'a'));
156 if (c >= 'A' && c <= 'F') return static_cast<uint8_t>(10 + (c - 'A'));
157 throw std::invalid_argument("Flow_Color received invalid hex digit.");
158 };
159
160 const auto decodeHexByte = [&](std::size_t index) -> float {
161 const uint8_t hi = decodeHexNibble(hexRgba[index]);
162 const uint8_t lo = decodeHexNibble(hexRgba[index + 1]);
163 return static_cast<float>((hi << 4) | lo);
164 };
165
166 return Clay_Color{
167 decodeHexByte(1),
168 decodeHexByte(3),
169 decodeHexByte(5),
170 decodeHexByte(7),
171 };

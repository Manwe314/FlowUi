207 static void onHovered(InteractionContext& context) {
208 if (isEnabled(context.params)) {
209 requestCursor(context, false);
210 }

217 static void onPressed(InteractionContext& context) {
218 auto& state = context.state();
219 if (!isEnabled(context.params)) {
220 state.isDragging = false;
221 return;
222 }
223
224 const detail::slider::Bounds bounds = resolveBounds(context.params);
225 const FrameInput& input = context.uiManager.getCurrentFrameInput();
226 state.isDragging = true;
227 state.pointerAtPress = pointerPosition(context.params.axis, input);
228 state.valueAtPress = detail::slider::snapValue(
229 *context.params.value,
230 bounds,
231 context.params.roundingStep);
232
233 invokeIfBound(context, context.params.onBegin);
234 if (context.params.pressBehavior == SliderPressBehavior::JumpToPointer) {
235 updateFromPointer(context, bounds);
236 }
237 }
238
243 static void runLogic(InteractionContext& context) {
244 auto& state = context.state();
245 if (!state.isDragging) {
246 return;
247 }
248 if (!isEnabled(context.params)) {
249 state.isDragging = false;
250 return;
251 }
252
253 const FrameInput& input = context.uiManager.getCurrentFrameInput();
254 if (!input.mouseDown[0]) {
255 state.isDragging = false;
256 invokeIfBound(context, context.params.onCommit);
257 return;
258 }
259
260 requestCursor(context, true);
261 updateFromPointer(context, resolveBounds(context.params));
262 }

268 static void buildElement(BuildContext& context) {
269 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().sliderTheme;
270 const Geometry geometry = resolveGeometry(context.params, theme);
271 const FSELSliderStateTheme appearance = resolveAppearance(context, theme);
272 const detail::slider::Bounds bounds = resolveBounds(context.params);
273 const double ratio = context.params.value
274 ? detail::slider::normalizedRatio(*context.params.value, bounds)
275 : 0.0;
276 const bool minimumAtLeading = valueDirection(context.params) > 0.0;
277 const float leadingRatio = minimumAtLeading
278 ? static_cast<float>(ratio)
279 : 1.0f - static_cast<float>(ratio);
280 const float travel = geometry.length - geometry.thumbLength;
281 const float thumbLeadingOffset = geometry.thumbLength * 0.5f +
282 leadingRatio * travel;
283 const float fillLength = static_cast<float>(ratio) * travel;
284
285 Clay_ElementDeclaration root{};
286 root.layout.sizing = {
287 .width = context.params.axis == SliderAxis::Horizontal
288 ? CLAY_SIZING_FIXED(geometry.length)
289 : CLAY_SIZING_FIXED(geometry.crossSize),
290 .height = context.params.axis == SliderAxis::Vertical
291 ? CLAY_SIZING_FIXED(geometry.length)
292 : CLAY_SIZING_FIXED(geometry.crossSize),
293 };
294
295 CLAY(context.clayID(), root) {
296 if (hasSliderVisualPart(
297 context.params.visualParts,
298 SliderVisualParts::Track)) {
299 emitTrack(context, geometry, appearance, theme);
300 }
301 if (fillLength > 0.0f && hasSliderVisualPart(
302 context.params.visualParts,
303 SliderVisualParts::Fill)) {
304 emitFill(
305 context,
306 geometry,
307 appearance,
308 theme,
309 fillLength,
310 minimumAtLeading);
311 }
312 if (hasSliderVisualPart(
313 context.params.visualParts,
314 SliderVisualParts::Thumb)) {
315 emitThumb(
316 context,
317 geometry,
318 appearance,
319 theme,
320 thumbLeadingOffset);
321 }
322 }
323 }

344 static Geometry resolveGeometry(
345 const SliderParameters& params,
346 const FSELSliderTheme& theme) {
347 Geometry geometry{};
348 geometry.trackThickness = std::max(
349 params.trackThickness.value_or(theme.trackThickness),
350 1.0f);
351 geometry.thumbLength = std::max(
352 params.thumbLength.value_or(theme.thumbLength),
353 1.0f);
354 geometry.thumbThickness = std::max(
355 params.thumbThickness.value_or(theme.thumbThickness),
356 1.0f);
357 geometry.length = std::max(
358 params.length.value_or(theme.length),
359 geometry.thumbLength + 1.0f);
360 geometry.crossSize = std::max({
361 params.hitThickness.value_or(theme.hitThickness),
362 geometry.trackThickness,
363 geometry.thumbThickness,
364 1.0f,
365 });
366 geometry.trackRoundness = detail::slider::clampRoundness(
367 params.trackRoundness.value_or(theme.trackRoundness));
368 geometry.thumbRoundness = detail::slider::clampRoundness(
369 params.thumbRoundness.value_or(theme.thumbRoundness));
370 return geometry;
371 }

373 static double valueDirection(const SliderParameters& params) {
374 double direction = params.axis == SliderAxis::Horizontal ? 1.0 : -1.0;
375 return params.inverted ? -direction : direction;

378 static float pointerPosition(SliderAxis axis, const FrameInput& input) {
379 return axis == SliderAxis::Horizontal ? input.mouseX : input.mouseY;
380 }

382 static double pointerRatio(
383 InteractionContext& context,
384 const Geometry& geometry) {
385 const Clay_ElementData rootData = Clay_GetElementData(context.clayID());
386 if (!rootData.found) {
387 return detail::slider::normalizedRatio(
388 *context.params.value,
389 resolveBounds(context.params));
390 }
391
392 const bool horizontal = context.params.axis == SliderAxis::Horizontal;
393 const float origin = horizontal
394 ? rootData.boundingBox.x
395 : rootData.boundingBox.y;
396 const float actualLength = horizontal
397 ? rootData.boundingBox.width
398 : rootData.boundingBox.height;
399 const float thumbLength = std::min(geometry.thumbLength, actualLength);
400 const float travel = std::max(actualLength - thumbLength, 1.0f);
401 const float pointer = pointerPosition(
402 context.params.axis,
403 context.uiManager.getCurrentFrameInput());
404 return std::clamp(
405 static_cast<double>((pointer - origin - thumbLength * 0.5f) / travel),
406 0.0,
407 1.0);
408 }

410 static void updateFromPointer(
411 InteractionContext& context,
412 detail::slider::Bounds bounds) {
413 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().sliderTheme;
414 const Geometry geometry = resolveGeometry(context.params, theme);
415 double ratio = 0.0;
416 if (context.params.pressBehavior == SliderPressBehavior::JumpToPointer) {
417 ratio = pointerRatio(context, geometry);
418 if (valueDirection(context.params) < 0.0) {
419 ratio = 1.0 - ratio;
420 }
421 } else {
422 const Clay_ElementData rootData = Clay_GetElementData(context.clayID());
423 const bool horizontal = context.params.axis == SliderAxis::Horizontal;
424 const float actualLength = rootData.found
425 ? (horizontal
426 ? rootData.boundingBox.width
427 : rootData.boundingBox.height)
428 : geometry.length;
429 const double travel = std::max(
430 static_cast<double>(actualLength - geometry.thumbLength),
431 1.0);
432 const float pointer = pointerPosition(
433 context.params.axis,
434 context.uiManager.getCurrentFrameInput());
435 const double deltaRatio =
436 static_cast<double>(pointer - context.state().pointerAtPress) /
437 travel * valueDirection(context.params);
438 ratio = detail::slider::normalizedRatio(
439 context.state().valueAtPress,
440 bounds) + deltaRatio;
441 }
442
443 const double nextValue = detail::slider::snapValue(
444 detail::slider::valueFromRatio(ratio, bounds),
445 bounds,
446 context.params.roundingStep);
447 if (nextValue == *context.params.value) {
448 return;
449 }
450 *context.params.value = nextValue;
451 invokeIfBound(context, context.params.onChanged);
452 }

488 static FSELSliderStateTheme resolveAppearance(
489 BuildContext& context,
490 const FSELSliderTheme& theme) {
491 if (!isEnabled(context.params)) {
492 return applyOverrides(theme.disabled, context.params.disabledOverrides);
493 }
494 if (context.state().isDragging) {
495 return applyOverrides(theme.dragging, context.params.draggingOverrides);

517 static void emitTrack(
518 BuildContext& context,
519 const Geometry& geometry,
520 const FSELSliderStateTheme& appearance,
521 const FSELSliderTheme& theme) {
522 const float travel = geometry.length - geometry.thumbLength;
523 Clay_ElementDeclaration track{};
524 track.layout.sizing = {
525 .width = context.params.axis == SliderAxis::Horizontal
526 ? CLAY_SIZING_FIXED(travel)
527 : CLAY_SIZING_FIXED(geometry.trackThickness),
528 .height = context.params.axis == SliderAxis::Vertical
529 ? CLAY_SIZING_FIXED(travel)
530 : CLAY_SIZING_FIXED(geometry.trackThickness),
531 };
532 track.backgroundColor = appearance.trackColor;
533 track.cornerRadius = CLAY_CORNER_RADIUS(
534 geometry.trackThickness * 0.5f * geometry.trackRoundness);
535 track.border = {
536 .color = appearance.trackBorderColor,
537 .width = context.params.trackBorderWidth.value_or(
538 theme.trackBorderWidth),
539 };
540 track.floating = centeredFloating();
541 CLAY(FlowUi::ValueOrThrow(context.clayID("track")), track);
542 }

544 static void emitFill(
545 BuildContext& context,
546 const Geometry& geometry,
547 const FSELSliderStateTheme& appearance,
548 const FSELSliderTheme& theme,
549 float fillLength,

552 fill.layout.sizing = {

555 : CLAY_SIZING_FIXED(geometry.trackThickness),

559 };
560 fill.backgroundColor = appearance.fillColor;
561 fill.cornerRadius = CLAY_CORNER_RADIUS(
562 geometry.trackThickness * 0.5f * geometry.trackRoundness);
563 fill.border = {
564 .color = appearance.fillBorderColor,
565 .width = context.params.fillBorderWidth.value_or(
566 theme.fillBorderWidth),
567 };
568 fill.floating.pointerCaptureMode = CLAY_POINTER_CAPTURE_MODE_PASSTHROUGH;
569 fill.floating.attachTo = CLAY_ATTACH_TO_PARENT;
570 fill.floating.clipTo = CLAY_CLIP_TO_ATTACHED_PARENT;
571
572 const float endpointOffset = geometry.thumbLength * 0.5f;
573 if (context.params.axis == SliderAxis::Horizontal) {
574 fill.floating.offset.x = minimumAtLeading

576 : -endpointOffset;
577 fill.floating.attachPoints = minimumAtLeading
578 ? Clay_FloatingAttachPoints{
579 .element = CLAY_ATTACH_POINT_LEFT_CENTER,
580 .parent = CLAY_ATTACH_POINT_LEFT_CENTER,
581 }
582 : Clay_FloatingAttachPoints{
583 .element = CLAY_ATTACH_POINT_RIGHT_CENTER,
584 .parent = CLAY_ATTACH_POINT_RIGHT_CENTER,
585 };
586 } else {
587 fill.floating.offset.y = minimumAtLeading
588 ? endpointOffset
589 : -endpointOffset;
590 fill.floating.attachPoints = minimumAtLeading
591 ? Clay_FloatingAttachPoints{
592 .element = CLAY_ATTACH_POINT_CENTER_TOP,
593 .parent = CLAY_ATTACH_POINT_CENTER_TOP,
594 }

596 .element = CLAY_ATTACH_POINT_CENTER_BOTTOM,
597 .parent = CLAY_ATTACH_POINT_CENTER_BOTTOM,
598 };
599 }
600 CLAY(FlowUi::ValueOrThrow(context.clayID("fill")), fill);
601 }

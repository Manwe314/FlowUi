36template<NumericValueType T>
37[[nodiscard]] std::string sanitize(
38 std::string_view text,
39 bool allowScientificInput) {
40 std::string result;
41 result.reserve(text.size());
42 bool hasDecimalPoint = false;
43 bool hasExponent = false;
44 bool hasMantissaDigit = false;
45 bool exponentMayTakeSign = false;
46
47 for (const char character : text) {
48 if (character >= '0' && character <= '9') {
49 result.push_back(character);
50 if (!hasExponent) {
51 hasMantissaDigit = true;
52 }
53 exponentMayTakeSign = false;
54 continue;
55 }
56 if (character == '-' && result.empty() &&
57 (std::signed_integral<T> || std::floating_point<T>)) {
58 result.push_back(character);
59 continue;
60 }
61 if constexpr (std::floating_point<T>) {
62 if (character == '.' && !hasDecimalPoint && !hasExponent) {
63 result.push_back(character);
64 hasDecimalPoint = true;
65 continue;
66 }
67 if (allowScientificInput && (character == 'e' || character == 'E') &&
68 !hasExponent && hasMantissaDigit) {
69 result.push_back(character);
70 hasExponent = true;
71 exponentMayTakeSign = true;
72 continue;
73 }
74 if (allowScientificInput && (character == '+' || character == '-') &&
75 exponentMayTakeSign) {
76 result.push_back(character);
77 exponentMayTakeSign = false;
78 continue;
79 }
80 }
81 }
82 return result;
83}

85inline bool isIncompleteFloatingText(std::string_view text) {
86 if (text.empty() || text == "-" || text == "." || text == "-.") {
87 return true;
88 }
89 const char last = text.back();
90 if (last == 'e' || last == 'E') {
91 return true;
92 }
93 if ((last == '+' || last == '-') && text.size() >= 2u) {
94 const char previous = text[text.size() - 2u];
95 return previous == 'e' || previous == 'E';
96 }
97 return false;
98}

100template<NumericValueType T>
101[[nodiscard]] ParseResult<T> parse(std::string_view text) {
102 ParseResult<T> result{};
103 result.negative = !text.empty() && text.front() == '-';
104 if (text.empty() || text == "-" ||
105 (std::floating_point<T> && isIncompleteFloatingText(text))) {
106 result.status = ParseStatus::Incomplete;
107 return result;
108 }
109
110 T value{};
111 const char* const begin = text.data();
112 const char* const end = begin + text.size();
113 std::from_chars_result parsed{};
114 if constexpr (std::integral<T>) {
115 parsed = std::from_chars(begin, end, value, 10);
116 } else {
117 parsed = std::from_chars(begin, end, value, std::chars_format::general);
118 }
119 if (parsed.ec == std::errc::result_out_of_range) {
120 result.status = ParseStatus::Overflow;
121 return result;
122 }
123 if (parsed.ec != std::errc{} || parsed.ptr != end ||
124 !detail::numeric::isFinite(value)) {
125 result.status = ParseStatus::Invalid;
126 return result;
127 }
128 result.status = ParseStatus::Complete;
129 result.value = value;
130 return result;
131}

133template<NumericValueType T>
134[[nodiscard]] ClassifiedValue<T> classify(
135 const ParseResult<T>& parsed,
136 detail::numeric::Bounds<T> bounds) {
137 if (!bounds.valid) {
138 return {.status = NumericTextStatus::ConfigurationError};
139 }
140 switch (parsed.status) {
141 case ParseStatus::Incomplete:
142 case ParseStatus::Invalid:
143 return {.status = NumericTextStatus::Incomplete};
144 case ParseStatus::Overflow:
145 return {
146 .status = NumericTextStatus::Overflow,
147 .candidate = parsed.negative ? bounds.lower : bounds.upper,
148 };
149 case ParseStatus::Complete:
150 break;
151 }
152 if (parsed.value < bounds.lower) {
153 return {
154 .status = NumericTextStatus::BelowMinimum,
155 .candidate = bounds.lower,
156 };
157 }
158 if (parsed.value > bounds.upper) {
159 return {
160 .status = NumericTextStatus::AboveMaximum,
161 .candidate = bounds.upper,
162 };
163 }
164 return {
165 .status = NumericTextStatus::Complete,
166 .candidate = parsed.value,
167 };

191inline void trimTrailingZeros(FormattedValue& formatted) {
192 const std::string_view text = formatted.view();
193 const size_t exponent = text.find_first_of("eE");
194 const size_t mantissaEnd = exponent == std::string_view::npos
195 ? text.size()
196 : exponent;
197 const size_t decimal = text.substr(0, mantissaEnd).find('.');
198 if (decimal == std::string_view::npos) {
199 return;
200 }
201
202 size_t trimmedEnd = mantissaEnd;
203 while (trimmedEnd > decimal + 1u &&
204 formatted.bytes[trimmedEnd - 1u] == '0') {
205 --trimmedEnd;
206 }
207 if (trimmedEnd == decimal + 1u) {
208 trimmedEnd = decimal;
209 }
210 if (trimmedEnd == mantissaEnd) {
211 return;
212 }
213 if (exponent != std::string_view::npos) {
214 const size_t exponentBytes = formatted.size - exponent;
215 std::memmove(
216 formatted.bytes.data() + trimmedEnd,
217 formatted.bytes.data() + exponent,
218 exponentBytes);
219 formatted.size = trimmedEnd + exponentBytes;
220 } else {
221 formatted.size = trimmedEnd;
222 }
223}

225template<NumericValueType T>
226[[nodiscard]] FormattedValue format(
227 T value,
228 const NumericFormatOptions& options = {}) {
229 FormattedValue result{};
230 if constexpr (std::floating_point<T>) {
231 if (options.canonicalizeNegativeZero && value == T{}) {
232 value = T{};
233 }
234 }
235
236 std::to_chars_result formatted{};
237 if constexpr (std::integral<T>) {
238 formatted = std::to_chars(
239 result.bytes.data(),
240 result.bytes.data() + result.bytes.size(),
241 value,
242 10);
243 } else {
244 const std::chars_format notation = toCharsFormat(options.notation);
245 if (options.precision.has_value() && *options.precision >= 0) {
246 formatted = std::to_chars(
247 result.bytes.data(),
248 result.bytes.data() + result.bytes.size(),
249 value,
250 notation,
251 *options.precision);
252 } else {
253 formatted = std::to_chars(
254 result.bytes.data(),
255 result.bytes.data() + result.bytes.size(),
256 value,
257 notation);
258 }
259 }
260 if (formatted.ec != std::errc{}) {
261 return result;
262 }
263 result.size = static_cast<size_t>(formatted.ptr - result.bytes.data());
264 if constexpr (std::floating_point<T>) {
265 if (options.trimTrailingZeros) {
266 trimTrailingZeros(result);
267 }
268 }
269 return result;

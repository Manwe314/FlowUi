85template <class Container>
86void append_capacity_demand(MemorySampleSink& sink, FlowUi::detail::InitialCapacityTarget target,
87 const Container& container,
88 const ::FlowUi::detail::MemoryDemandPeaks& peaks,
89 WindowId window = InvalidWindowId, uint64_t sentinel = 0) noexcept {
90 DevContainerMemoryAccumulator estimate{};
91 if constexpr (requires { container.bucket_count(); })
92 estimate.addNodeContainer(container);
93 else
94 estimate.add(container);
95 const auto count = container.size() > sentinel ? container.size() - sentinel : 0;
96 auto sample = estimate.sample(
97 FlowUi::detail::initial_capacity_definitions[static_cast<size_t>(target)].id(), window);
98 sample.objectCount = peaks.consume(target, count);
99 sample.peakObjectCount = sample.objectCount;
100 sample.flags = sample.flags | MemorySampleFlag::TuningAggregate;
101 sample.value_fields =
102 uint16_t(MemoryValueField::Objects) | uint16_t(MemoryValueField::Capacity) |
103 uint16_t(MemoryValueField::Backing) | uint16_t(MemoryValueField::LogicalLive);
104 if (FlowUi::detail::initial_capacity_definitions[static_cast<size_t>(target)].unit ==
105 FlowUi::detail::InitialCapacityUnit::Bytes) {
106 sample.peakLogicalBytes = sample.objectCount;
107 sample.value_fields |= uint16_t(MemoryValueField::PeakLogical);
108 }
109 (void)sink.append(sample);
110}

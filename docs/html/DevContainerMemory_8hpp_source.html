69template <class Container>
70void append_capacity_demand(MemorySampleSink& sink, FlowUi::detail::InitialCapacityTarget target,
71 const Container& container,
72 const ::FlowUi::detail::MemoryDemandPeaks& peaks,
73 WindowId window = InvalidWindowId, uint64_t sentinel = 0) noexcept {
74 DevContainerMemoryAccumulator estimate{};
75 if constexpr (requires { container.bucket_count(); })
76 estimate.addNodeContainer(container);
77 else
78 estimate.add(container);
79 const auto count = container.size() > sentinel ? container.size() - sentinel : 0;
80 auto sample = estimate.sample(
81 FlowUi::detail::initial_capacity_definitions[static_cast<size_t>(target)].id(), window);
82 sample.objectCount = peaks.consume(target, count);
83 sample.peakObjectCount = sample.objectCount;
84 sample.flags = sample.flags | MemorySampleFlag::TuningAggregate;
85 sample.value_fields =
86 uint16_t(MemoryValueField::Objects) | uint16_t(MemoryValueField::Capacity) |
87 uint16_t(MemoryValueField::Backing) | uint16_t(MemoryValueField::LogicalLive);
88 if (FlowUi::detail::initial_capacity_definitions[static_cast<size_t>(target)].unit ==
89 FlowUi::detail::InitialCapacityUnit::Bytes) {
90 sample.peakLogicalBytes = sample.objectCount;
91 sample.value_fields |= uint16_t(MemoryValueField::PeakLogical);
92 }
93 (void)sink.append(sample);
94}

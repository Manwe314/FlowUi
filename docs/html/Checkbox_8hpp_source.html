203 static FSELCheckboxStateTheme resolveAppearance(
204 BuildContext& context,
205 const FSELCheckboxTheme& theme) {
206 const FSELCheckboxValueTheme& valueTheme = context.params.isChecked
207 ? theme.checked
208 : theme.unchecked;
209 const CheckboxValueOverrides& overrides = context.params.isChecked
210 ? context.params.checkedOverrides
211 : context.params.uncheckedOverrides;
212
213 switch (detail::boolean_control::resolveVisualState(
214 context,
215 context.params.onToggle)) {
216 case detail::boolean_control::VisualState::Idle:
217 return applyOverrides(valueTheme.idle, overrides.idle);
218 case detail::boolean_control::VisualState::Hovered:
219 return applyOverrides(valueTheme.hovered, overrides.hovered);
220 case detail::boolean_control::VisualState::Pressed:
221 return applyOverrides(valueTheme.pressed, overrides.pressed);
222 case detail::boolean_control::VisualState::Disabled:
223 return applyOverrides(valueTheme.disabled, overrides.disabled);
224 }
225 return applyOverrides(valueTheme.idle, overrides.idle);
226 }

204[[nodiscard]] constexpr MemorySourceId
205initial_capacity_parent(const FlowUi::detail::InitialCapacityDefinition& definition) noexcept {
206 const auto key = definition.key;
207 if (key.starts_with("flowui.tuning.storage.") || key.starts_with("flowui.tuning.allocator."))
208 return kStorageMetadata.id;
209 if (key.starts_with("flowui.tuning.input_fields."))
210 return kInputFields.id;
211 if (key.starts_with("flowui.tuning.elements."))
212 return kElements.id;
213 if (key.starts_with("flowui.tuning.popups."))
214 return kPopups.id;
215 if (key.starts_with("flowui.tuning.shortcuts."))
216 return kShortcuts.id;
217 if (key.starts_with("flowui.tuning.actions."))
218 return kActions.id;
219 if (key.starts_with("flowui.tuning.fonts."))
220 return kFonts.id;
221 if (key.starts_with("flowui.tuning.icons."))
222 return kIcons.id;
223 if (key.starts_with("flowui.tuning.themes."))
224 return kThemes.id;
225 if (key.starts_with("flowui.tuning.viewports."))
226 return kViewports.id;
227 if (key.starts_with("flowui.tuning.development."))
228 return kDevTreeCapture.id;
229 if (key.starts_with("flowui.tuning.ui."))
230 return kUiLayout.id;
231 return kManagers.id;
232}

145 static void buildElement(BuildContext& context) {
146 const FSELTextFieldTheme& theme =
147 FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().textInputTheme;
148 const Clay_Padding padding = context.params.padding.value_or(theme.padding);
149 const Clay_BorderWidth borderWidth =
150 context.params.borderWidth.value_or(theme.borderWidth);
151 const Clay_CornerRadius cornerRadius =
152 context.params.cornerRadius.value_or(theme.cornerRadius);
153 const Clay_ElementId contentId = context.uiManager.toClayEID(
154 FlowUi::ValueOrThrow(context.part(Parts::content)));
155 const Clay_ElementId textId = context.uiManager.toClayEID(
156 FlowUi::ValueOrThrow(context.part(Parts::text)));
157 const Clay_ElementId placeholderId = context.uiManager.toClayEID(
158 FlowUi::ValueOrThrow(context.part(Parts::placeholder)));
159 const Clay_Dimensions viewport = detail::text_field::resolveViewport(
160 contentId,
161 theme,
162 padding,
163 context.params.viewportWidth,
164 context.params.viewportHeight);
165
166 Clay_TextElementConfig textConfig{};
167 textConfig.fontId = context.uiManager.resolveFont(
168 context.params.fontFamily.value_or(theme.fontFamily),
169 context.params.fontWeight.value_or(theme.fontWeight),
170 context.params.fontStyle.value_or(theme.fontStyle));
171 textConfig.fontSize = context.params.fontSize.value_or(theme.fontSize);
172 textConfig.letterSpacing = context.params.letterSpacing.value_or(
173 theme.letterSpacing);
174 textConfig.wrapMode = CLAY_TEXT_WRAP_NONE;
175 textConfig.textAlignment = CLAY_TEXT_ALIGN_LEFT;
176
177 const detail::text_field::BuildResult buildResult =
178 detail::text_field::requestAndSynchronize(
179 context,
180 context.params,
181 TextFieldMode::SingleLine,
182 textId,
183 contentId,
184 TextLayoutDescriptor{
185 .fontId = textConfig.fontId,
186 .fontSize = textConfig.fontSize,
187 .letterSpacing = textConfig.letterSpacing,
188 .viewportWidth = viewport.width,
189 .viewportHeight = viewport.height,
190 .tabWidth = theme.tabWidth,
191 .floatingZIndex = context.params.floatingZIndex.value_or(0),
192 },
193 detail::text_field::resolveOverlayStyle(theme, context.params.caret),
194 false,
195 context.params.clearFocusOnSubmit);
196 const FieldQueryResult& field = buildResult.field;
197
198 const bool enabled = context.params.enabled && context.params.value != nullptr;
199 const bool hovered = context.uiManager.getPreviousFramesInteraction()
200 .isHovered(context.clayID());
201 const FSELTextFieldStateTheme appearance =
202 detail::text_field::resolveAppearance(
203 theme,
204 context.params,
205 enabled,
206 field.hasPrimaryCaret,
207 hovered);
208 textConfig.textColor = appearance.textColor;
209 Clay_TextElementConfig placeholderConfig = textConfig;
210 placeholderConfig.textColor = appearance.placeholderColor;
211
212 const Clay_ElementDeclaration root =
213 detail::text_field::makeRootDeclaration(
214 detail::text_field::resolveSizing(context.params, theme),
215 padding,
216 borderWidth,
217 cornerRadius,
218 appearance);
219 const Clay_ElementDeclaration content =
220 detail::text_field::makeContentDeclaration();
221
222 CLAY(context.clayID(), root) {
223 CLAY(contentId, content) {
224 if (!field.visibleLines.empty()) {
225 const VisibleTextLine& line = field.visibleLines.front();
226 auto line_declaration = line.declaration;
227 if (context.params.centerText) {
228 const auto previous_text = Clay_GetElementData(textId);
229 const auto previous_content = Clay_GetElementData(contentId);
230 const bool fits =
231 !previous_text.found || !previous_content.found ||
232 previous_text.boundingBox.width <=
233 previous_content.boundingBox.width - padding.left - padding.right;
234 line_declaration.floating.attachPoints =
235 fits ? Clay_FloatingAttachPoints{CLAY_ATTACH_POINT_CENTER_CENTER,
236 CLAY_ATTACH_POINT_CENTER_CENTER}
237 : Clay_FloatingAttachPoints{CLAY_ATTACH_POINT_LEFT_CENTER,
238 CLAY_ATTACH_POINT_LEFT_CENTER};
239 if (fits)
240 line_declaration.floating.offset.x = 0;
241 line_declaration.floating.offset.y = 0;
242 }
243 CLAY(textId, line_declaration) {
244 CLAY_TEXT(context.uiManager.toClayString(line.text),
245 CLAY_TEXT_CONFIG(textConfig));
246 }
247 }
248 if (field.text.empty() && !context.params.placeholder.empty()) {
249 auto placeholder_declaration = detail::text_field::makePlaceholderDeclaration(
250 context.params.floatingZIndex.value_or(0));
251 if (context.params.centerText) {
252 placeholder_declaration.floating.attachPoints = {
253 CLAY_ATTACH_POINT_CENTER_CENTER, CLAY_ATTACH_POINT_CENTER_CENTER};
254 placeholder_declaration.floating.offset = {};
255 }
256 CLAY(placeholderId, placeholder_declaration) {
257 CLAY_TEXT(context.uiManager.toClayString(context.params.placeholder),
258 CLAY_TEXT_CONFIG(placeholderConfig));
259 }
260 }
261 }
262 }
263
264 if (!field.visibleLines.empty()) {
265 const VisibleTextLine& line = field.visibleLines.front();
266 (void)context.uiManager.inputFields().submitTextSpan(
267 field.field,
268 FieldTextSpanSubmission{
269 .logicalRange = line.logicalRange,
270 .textElementId = textId,
271 .visualLineIndex = line.visualLineIndex,
272 });
273 }
274 detail::text_field::invokeActionsAfterBuild(
275 context,
276 context.params,
277 buildResult.notifications);
278 }

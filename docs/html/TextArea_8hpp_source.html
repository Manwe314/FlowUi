162 Clay_TextElementConfig textConfig{};
163 textConfig.fontId = context.uiManager.resolveFont(
164 context.params.fontFamily.value_or(theme.fontFamily),
165 context.params.fontWeight.value_or(theme.fontWeight),
166 context.params.fontStyle.value_or(theme.fontStyle));
167 textConfig.fontSize = context.params.fontSize.value_or(theme.fontSize);
168 textConfig.letterSpacing = context.params.letterSpacing.value_or(

171 textConfig.textAlignment = CLAY_TEXT_ALIGN_LEFT;

173 const detail::text_field::BuildResult buildResult =
174 detail::text_field::requestAndSynchronize(

178 textLayerId,
179 contentId,
180 TextLayoutDescriptor{

182 .fontSize = textConfig.fontSize,
183 .letterSpacing = textConfig.letterSpacing,
184 .viewportWidth = viewport.width,
185 .viewportHeight = viewport.height,
186 .tabWidth = context.params.tabWidth.value_or(theme.tabWidth),
187 },
188 detail::text_field::resolveOverlayStyle(theme, context.params.caret),
189 context.params.softWrap,
190 false);

192
193 const bool enabled = context.params.enabled && context.params.value != nullptr;
194 const bool hovered = context.uiManager.getPreviousFramesInteraction()
195 .isHovered(context.clayID());
196 const FSELTextFieldStateTheme appearance =
197 detail::text_field::resolveAppearance(
198 theme,
199 context.params,
200 enabled,
201 field.hasPrimaryCaret,
202 hovered);
203 textConfig.textColor = appearance.textColor;
204 Clay_TextElementConfig placeholderConfig = textConfig;
205 placeholderConfig.textColor = appearance.placeholderColor;
206
207 const Clay_ElementDeclaration root =
208 detail::text_field::makeRootDeclaration(
209 detail::text_field::resolveSizing(context.params, theme),
210 padding,
211 borderWidth,
212 cornerRadius,
213 appearance);
214 const Clay_ElementDeclaration content =
215 detail::text_field::makeContentDeclaration();
216 Clay_ElementDeclaration textLayer =
217 detail::text_field::makeContentDeclaration();
218
219 CLAY(context.clayID(), root) {
220 CLAY(contentId, content) {
221 CLAY(textLayerId, textLayer) {
222 for (const VisibleTextLine& line : field.visibleLines) {
223 const std::string lineName =
224 "line-" + std::to_string(line.visualLineIndex);
225 const Clay_ElementId lineId = FlowUi::ValueOrThrow(context.clayID(
226 RuntimeName(lineName)));
227 CLAY(lineId, line.declaration) {
228 CLAY_TEXT(
229 context.uiManager.toClayString(line.text),
230 CLAY_TEXT_CONFIG(textConfig));
231 }
232 (void)context.uiManager.inputFields().submitTextSpan(
233 field.field,
234 FieldTextSpanSubmission{
235 .logicalRange = line.logicalRange,
236 .textElementId = lineId,
237 .visualLineIndex = line.visualLineIndex,
238 });
239 }
240 }
241 if (field.text.empty() && !context.params.placeholder.empty()) {
242 CLAY(
243 placeholderId,
244 detail::text_field::makePlaceholderDeclaration()) {
245 CLAY_TEXT(
246 context.uiManager.toClayString(context.params.placeholder),
247 CLAY_TEXT_CONFIG(placeholderConfig));
248 }
249 }
250 }
251 }
252 detail::text_field::invokeActionsAfterBuild(
253 context,
254 context.params,
255 buildResult.notifications);
256 }

247 static void onPressed(InteractionContext& context) {
248 auto& state = context.state();
249 if (state.mode == DragValueInteractionMode::Editing) {
250 return;
251 }
252 if (!canStartInteraction(context.params)) {
253 state.mode = DragValueInteractionMode::Idle;
254 return;

257 const FrameInput& input = context.uiManager.getCurrentFrameInput();
258 state.mode = DragValueInteractionMode::Pending;

260 state.valueAtPress = *context.params.value;

267 static void onReleased(InteractionContext& context) {
268 auto& state = context.state();
269 if (state.mode != DragValueInteractionMode::Pending) {
270 return;
271 }
272 // A drag transitions out of Pending while the button is held. If the
273 // gesture is still pending at release, classify it as a click so small
274 // pointer jitter cannot prevent click-to-edit.
275 if (canEditText(context.params)) {
276 state.mode = DragValueInteractionMode::Editing;
277 state.forceTextSynchronization = true;
278 context.uiManager.inputFields().requestCaret(
279 context.id,
280 CaretRequestKind::SetPrimary);
281 } else {
282 state.mode = DragValueInteractionMode::Idle;
283 }
284 }

290 static void runLogic(InteractionContext& context) {
291 auto& state = context.state();
292 if (state.mode == DragValueInteractionMode::Idle ||
293 state.mode == DragValueInteractionMode::Editing) {
294 return;
295 }
296 if (!hasUsableBinding(context.params)) {
297 cancelGesture(context);
298 return;
299 }
300
301 const FrameInput& input = context.uiManager.getCurrentFrameInput();
302 const float horizontalDelta = input.mouseX - state.pointerXAtPress;
303 if (state.mode == DragValueInteractionMode::Pending) {
304 if (!input.mouseDown[0]) {
305 // Keep the press pending until onReleased classifies it as a
306 // click-to-edit gesture. Clearing it here races the release
307 // callback and makes every short click behave like a cancelled drag.
308 return;
309 }
310 if (!canDrag(context.params) ||
311 !detail::drag_value::crossedThreshold(
312 horizontalDelta,
313 context.params.dragThreshold)) {
314 return;
315 }
316 state.mode = DragValueInteractionMode::Dragging;
317 }
318
319 if (escapeIsDown(input)) {
320 cancelGesture(context);
321 return;
322 }
323 if (!canDrag(context.params)) {
324 cancelGesture(context);
325 return;
326 }
327
328 if (input.mouseDown[0]) {
329 requestDragCursor(context);
330 }
331 const detail::numeric::Bounds<T> bounds = resolveBounds(context.params);
332 const int64_t stepCount = detail::drag_value::stepCountForDelta(
333 horizontalDelta,
334 context.params.pixelsPerStep);
335 const T next = detail::numeric::offsetBySteps(
336 state.valueAtPress,
337 context.params.step,
338 stepCount,
339 bounds);
340 if (next != *context.params.value) {
341 detail::numeric_control::beginEdit<T>(context);
342 if (detail::numeric_control::applyValue<T>(context, next)) {
343 state.editSession.hasLastValidValue = true;
344 state.editSession.lastValidValue = next;
345 }
346 }
347 if (!input.mouseDown[0]) {
348 detail::numeric_control::commitEdit(context);
349 state.mode = DragValueInteractionMode::Idle;
350 state.forceTextSynchronization = true;
351 }
352 }

358 static void buildElement(BuildContext& context) {
359 const FSELDragValueTheme& dragTheme =
360 FlowUi::ValueOrThrow(context.uiManager.template theme<FSELTheme>()).get().dragValueTheme;
361 const FSELTextFieldTheme& theme = dragTheme.field;
362 auto& state = context.state();
363 auto& manager = context.uiManager.inputFields();
364 const detail::numeric::Bounds<T> bounds = resolveBounds(context.params);
365 const bool fieldEnabled = isBoundAndEnabled(context.params) && bounds.valid;
366 const bool textEditable = fieldEnabled && canEditText(context.params);
367
368 if (state.mode == DragValueInteractionMode::Editing && !textEditable) {
369 const bool hadFocus = state.wasFocused;
370 detail::numeric_control::cancelEdit<T>(context);
371 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
372 state.mode = DragValueInteractionMode::Idle;
373 state.wasFocused = false;
374 if (hadFocus) {
375 detail::numeric_control::invoke(context, context.params.onBlur);

377 }

380 const Clay_ElementId contentId = context.uiManager.toClayEID(
381 FlowUi::ValueOrThrow(context.part(Parts::content)));

383 FlowUi::ValueOrThrow(context.part(Parts::text)));

386 contentId,
387 theme,
388 padding,
389 context.params.viewportWidth,
390 context.params.viewportHeight);

398 textConfig.letterSpacing = context.params.letterSpacing.value_or(
399 theme.letterSpacing);

401 textConfig.textAlignment = CLAY_TEXT_ALIGN_LEFT;

404 const detail::numeric_text::FormattedValue formattedValue =

407 .initialText = formattedValue.view(),
408 .config = FieldConfig{
409 .maxBytes = context.params.maxBytes,
410 .mode = TextFieldMode::SingleLine,
411 .transactionDetail = context.params.transactionDetail,
412 .readOnly = !textEditable,

414 .allowArrowNavigation = true,

416 .layout = TextLayoutDescriptor{
417 .fontId = textConfig.fontId,
418 .fontSize = textConfig.fontSize,
419 .letterSpacing = textConfig.letterSpacing,
420 .viewportWidth = viewport.width,

426 context.params.caret),
427 .textElementId = editing && fieldEnabled ? textId : Clay_ElementId{},
428 .contentElementId = editing && fieldEnabled

432 FieldQueryResult field = manager.requestField(context.id, request);
433
434 const bool userTextChanged = editing && std::ranges::any_of(
435 field.transactions,
436 [](const FieldEditTransaction& transaction) {
437 return transaction.origin != EditOrigin::Programmatic;
438 });
439 if (textEditable && userTextChanged) {
440 const std::string fieldText = FlowUi::ValueOrThrow(field.text.copy());
441 const std::string sanitized = detail::numeric_text::sanitize<T>(
442 fieldText,
443 context.params.format.allowScientificInput);
444 if (sanitized != fieldText) {
445 (void)manager.replaceText(context.id, sanitized, true);
446 field = manager.requestField(context.id, request);
447 }
448 }
449
450 bool focused = editing && fieldEnabled && field.hasPrimaryCaret;
451 if (state.forceTextSynchronization ||
452 (!editing && !userTextChanged && !state.editSession.active &&
453 !detail::text_field::textEquals(
454 field.text,
455 formattedValue.view()))) {
456 (void)manager.replaceText(
457 context.id,
458 formattedValue.view(),
459 focused);
460 field = manager.requestField(context.id, request);
461 focused = editing && fieldEnabled && field.hasPrimaryCaret;
462 state.forceTextSynchronization = false;
463 }
464
465 const detail::numeric_text::ClassifiedValue<T> classified =
466 detail::numeric_text::classify(
467 detail::numeric_text::parse<T>(FlowUi::ValueOrThrow(field.text.copy())),
468 bounds);
469 const std::optional<T> candidate = classified.candidate;
470 state.textStatus = classified.status;
471 if (!bounds.valid || !detail::numeric::isFinite(currentValue)) {
472 state.textStatus = NumericTextStatus::ConfigurationError;
473 }
474
475 if (userTextChanged && textEditable) {
476 detail::numeric_control::beginEdit<T>(context);
477 if (candidate.has_value()) {
478 state.editSession.hasLastValidValue = true;
479 state.editSession.lastValidValue = *candidate;
480 if (context.params.syncPolicy == NumericTextSyncPolicy::Live) {
481 (void)detail::numeric_control::applyValue<T>(
482 context,
483 *candidate);
484 }
485 }
486 }

488 const bool cancelled = editing && detail::text_field::hasCommand(
489 field,
490 FieldCommandRequest::Cancel);
491 const bool submitted = editing && detail::text_field::hasCommand(
492 field,
493 FieldCommandRequest::Submit);
494 const bool blurred = editing && state.wasFocused && !focused;
495 if (editing && !state.wasFocused && focused) {
496 detail::numeric_control::invoke(context, context.params.onFocus);
497 }
498
499 if (cancelled) {
500 detail::numeric_control::cancelEdit<T>(context);
501 const detail::numeric_text::FormattedValue restored =
502 detail::numeric_text::format(
503 context.params.value ? *context.params.value : T{},
504 context.params.format);
505 (void)manager.replaceText(context.id, restored.view(), true);
506 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
507 field = manager.requestField(context.id, request);
508 focused = false;
509 state.mode = DragValueInteractionMode::Idle;
510 } else if (submitted || blurred) {
511 detail::numeric_control::commitTextEdit<T>(
512 context,
513 candidate,
514 context.params.syncPolicy);
515 const detail::numeric_text::FormattedValue committed =
516 detail::numeric_text::format(
517 context.params.value ? *context.params.value : T{},
518 context.params.format);
519 (void)manager.replaceText(context.id, committed.view(), focused);
520 field = manager.requestField(context.id, request);
521 if (submitted) {
522 detail::numeric_control::invoke(context, context.params.onSubmit);
523 }
524 if (blurred || context.params.clearFocusOnSubmit) {
525 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
526 focused = false;
527 state.mode = DragValueInteractionMode::Idle;
528 }
529 }
530
531 if ((submitted && context.params.clearFocusOnSubmit) ||
532 blurred || cancelled) {
533 detail::numeric_control::invoke(context, context.params.onBlur);
534 }
535 if (editing && detail::text_field::hasCommand(
536 field,
537 FieldCommandRequest::Undo)) {
538 detail::numeric_control::invoke(
539 context,
540 context.params.onUndoRequested);
541 }
542 if (editing && detail::text_field::hasCommand(
543 field,
544 FieldCommandRequest::Redo)) {
545 detail::numeric_control::invoke(
546 context,
547 context.params.onRedoRequested);
548 }
549
550 if (state.mode == DragValueInteractionMode::Editing &&
551 context.params.boundsPolicy ==
552 NumericTextBoundsPolicy::ClampTextImmediately &&
553 candidate.has_value() &&
554 state.textStatus != NumericTextStatus::Complete) {
555 const detail::numeric_text::FormattedValue clamped =
556 detail::numeric_text::format(
557 *candidate,
558 context.params.format);
559 (void)manager.replaceText(context.id, clamped.view(), focused);
560 field = manager.requestField(context.id, request);
561 state.textStatus = NumericTextStatus::Complete;
562 }
563
564 const detail::numeric_text::FormattedValue displayValue =
565 detail::numeric_text::format(
566 context.params.value ? *context.params.value : T{},
567 context.params.format);
568 if (state.mode != DragValueInteractionMode::Editing) {
569 state.textStatus = detail::numeric_text::classify(
570 detail::numeric_text::parse<T>(displayValue.view()),
571 bounds).status;
572 if (!bounds.valid || (context.params.value &&
573 !detail::numeric::isFinite(*context.params.value))) {
574 state.textStatus = NumericTextStatus::ConfigurationError;
575 }
576 }
577
578 state.wasFocused = focused;
579 drawField(
580 context,
581 field,
582 displayValue.view(),
583 textConfig,
584 padding,
585 contentId,
586 textId,
587 isBoundAndEnabled(context.params),
588 focused,
589 dragTheme);

646 static FSELTextFieldStateTheme resolveAppearance(
647 BuildContext& context,
648 const FSELDragValueTheme& theme,
649 bool enabled,
650 bool focused) {
651 if (!enabled) {
652 return detail::text_field::applyOverrides(
653 theme.field.disabled,
654 context.params.disabledOverrides);
655 }
656 const bool dragConfigurationValid =
657 detail::numeric::validStep(context.params.step) &&
658 detail::drag_value::validPixelsPerStep(
659 context.params.pixelsPerStep) &&
660 std::isfinite(context.params.dragThreshold);
661 if (!context.params.valid || !resolveBounds(context.params).valid ||
662 !dragConfigurationValid ||
663 context.state().textStatus != NumericTextStatus::Complete) {
664 return detail::text_field::applyOverrides(
665 theme.field.invalid,
666 context.params.invalidOverrides);
667 }
668 if (context.state().mode == DragValueInteractionMode::Dragging) {
669 return detail::text_field::applyOverrides(
670 theme.dragging,
671 context.params.draggingOverrides);
672 }
673 if (focused) {
674 return detail::text_field::applyOverrides(
675 theme.field.focused,
676 context.params.focusedOverrides);
677 }
678 if (context.params.readOnly) {
679 return detail::text_field::applyOverrides(
680 theme.field.readOnly,
681 context.params.readOnlyOverrides);
682 }
683 if (context.uiManager.getPreviousFramesInteraction().isHovered(
684 context.clayID())) {
685 return detail::text_field::applyOverrides(
686 theme.field.hovered,
687 context.params.hoveredOverrides);
688 }
689 return detail::text_field::applyOverrides(
690 theme.field.idle,
691 context.params.idleOverrides);
692 }

694 static void drawField(
695 BuildContext& context,
696 const FieldQueryResult& field,
697 std::string_view displayText,
698 Clay_TextElementConfig textConfig,
699 Clay_Padding padding,
700 Clay_ElementId contentId,
701 Clay_ElementId textId,
702 bool enabled,
703 bool focused,
704 const FSELDragValueTheme& dragTheme) {
705 const FSELTextFieldTheme& theme = dragTheme.field;
706 const FSELTextFieldStateTheme appearance = resolveAppearance(
707 context,
708 dragTheme,
709 enabled,
710 focused);
711 textConfig.textColor = appearance.textColor;
712
713 Clay_ElementDeclaration root{};
714 root.layout.sizing = context.params.sizing.value_or(Clay_Sizing{
715 .width = CLAY_SIZING_FIXED(std::max(theme.width, 1.0f)),
716 .height = CLAY_SIZING_FIXED(std::max(theme.height, 1.0f)),
717 });
718 // Keep the content box itself inset. Editing lines are floating children
719 // of that box, so padding the content element would only affect the
720 // normal-flow display text and make it jump when editing begins.
721 root.layout.padding = padding;
722 root.layout.childAlignment = {
723 .x = CLAY_ALIGN_X_LEFT,
724 .y = CLAY_ALIGN_Y_TOP,
725 };
726 root.backgroundColor = appearance.backgroundColor;
727 root.cornerRadius = context.params.cornerRadius.value_or(
728 theme.cornerRadius);
729 root.clip = {.horizontal = true, .vertical = true};
730 root.border = {
731 .color = appearance.borderColor,
732 .width = context.params.borderWidth.value_or(theme.borderWidth),
733 };
734
735 const Clay_ElementDeclaration content =
736 detail::text_field::makeContentDeclaration();
737 const bool editing =
738 context.state().mode == DragValueInteractionMode::Editing;
739 CLAY(context.clayID(), root) {
740 CLAY(contentId, content) {
741 if (editing && !field.visibleLines.empty()) {
742 const VisibleTextLine& line = field.visibleLines.front();
743 CLAY(textId, line.declaration) {
744 CLAY_TEXT(
745 context.uiManager.toClayString(line.text),
746 CLAY_TEXT_CONFIG(textConfig));
747 }
748 } else if (!editing) {
749 CLAY(textId, {}) {
750 CLAY_TEXT(
751 context.uiManager.toClayString(displayText),
752 CLAY_TEXT_CONFIG(textConfig));
753 }
754 }
755 }
756 }
757
758 if (editing && !field.visibleLines.empty()) {
759 const VisibleTextLine& line = field.visibleLines.front();
760 (void)context.uiManager.inputFields().submitTextSpan(
761 field.field,
762 FieldTextSpanSubmission{
763 .logicalRange = line.logicalRange,
764 .textElementId = textId,
765 .visualLineIndex = line.visualLineIndex,
766 });
767 }

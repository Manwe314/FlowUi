15/* Pure axis math shared by scrollable FSEL surfaces and indicators. */
16[[nodiscard]] inline ThumbGeometry calculate(
17 float viewportExtent,
18 float contentExtent,
19 float scrollOffset,
20 float trackExtent,
21 float minimumThumbExtent) noexcept {
22 if (!std::isfinite(viewportExtent) || !std::isfinite(contentExtent) ||
23 !std::isfinite(scrollOffset) || !std::isfinite(trackExtent) ||
24 !std::isfinite(minimumThumbExtent) || viewportExtent <= 0.0f ||
25 contentExtent <= viewportExtent || trackExtent <= 0.0f) {
26 return {};
27 }
28
29 ThumbGeometry result{};
30 result.visible = true;
31 result.maximumScroll = contentExtent - viewportExtent;
32 result.extent = std::clamp(
33 trackExtent * viewportExtent / contentExtent,
34 std::clamp(minimumThumbExtent, 0.0f, trackExtent),
35 trackExtent);
36 const float normalizedScroll = std::clamp(
37 -scrollOffset / result.maximumScroll,
38 0.0f,
39 1.0f);
40 result.offset = normalizedScroll * (trackExtent - result.extent);
41 return result;
42}

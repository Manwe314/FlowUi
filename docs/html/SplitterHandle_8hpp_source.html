108 static void onHovered(InteractionContext& context) {

118 static void onPressed(InteractionContext& context) {
119 auto& state = context.state();
120 if (!context.params.targetExtent) {
121 state.isDragging = false;
122 return;
123 }
124
125 const FrameInput& input = context.uiManager.getCurrentFrameInput();
126 state.extentAtPress = *context.params.targetExtent;
127 state.pointerAtPress = pointerPosition(context.params.axis, input);
128 state.isDragging = true;
129 }

135 static void runLogic(InteractionContext& context) {
136 auto& state = context.state();
137 if (!state.isDragging) {
138 return;
139 }
140 if (!context.params.targetExtent) {
141 state.isDragging = false;
142 return;
143 }
144
145 const FrameInput& input = context.uiManager.getCurrentFrameInput();
146 if (!input.mouseDown[0]) {
147 state.isDragging = false;
148 return;
149 }
150
151 requestResizeCursor(context);
152 const float delta =
153 pointerPosition(context.params.axis, input) - state.pointerAtPress;
154 const float signedDelta =
155 context.params.position == SplitterPosition::Trailing ? delta : -delta;
156 const auto [minimumExtent, maximumExtent] = std::minmax(
157 context.params.minExtent,
158 context.params.maxExtent);
159 *context.params.targetExtent = std::clamp(
160 state.extentAtPress + signedDelta,
161 minimumExtent,
162 maximumExtent);
163 }

169 static void buildElement(BuildContext& context) {
170 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get();
171 const auto& splitterTheme = theme.splitterHandleTheme;
172 const auto& state = context.state();
173 const float thickness = std::max(context.params.thickness, 1.0f);
174 const float hitThickness = std::max(context.params.hitThickness, thickness);
175 const Clay_ElementId rootId = context.clayID();
176 const bool isHovered = context.params.targetExtent &&
177 context.uiManager.getPreviousFramesInteraction().isHovered(rootId);
178
179 Clay_ElementDeclaration root{};
180 root.layout.sizing = {
181 .width = context.params.axis == SplitterAxis::Horizontal
182 ? CLAY_SIZING_FIXED(hitThickness)
183 : CLAY_SIZING_GROW(0),
184 .height = context.params.axis == SplitterAxis::Vertical
185 ? CLAY_SIZING_FIXED(hitThickness)
186 : CLAY_SIZING_GROW(0),
187 };
188 root.layout.childAlignment = {
189 .x = CLAY_ALIGN_X_CENTER,
190 .y = CLAY_ALIGN_Y_CENTER,
191 };
192 const bool separate_surfaces = context.params.leadingSurfaceColor.has_value() ||
193 context.params.trailingSurfaceColor.has_value();
194 root.layout.layoutDirection = context.params.axis == SplitterAxis::Horizontal
195 ? CLAY_LEFT_TO_RIGHT
196 : CLAY_TOP_TO_BOTTOM;
197 if (state.isDragging && !separate_surfaces) {
198 root.backgroundColor =
199 context.params.draggingColor.value_or(splitterTheme.draggingColor);
200 } else if (isHovered && !separate_surfaces) {
201 root.backgroundColor = context.params.hoverColor.value_or(splitterTheme.hoverColor);
202 }
203
204 Clay_ElementDeclaration visual{};
205 visual.layout.sizing = {
206 .width = context.params.axis == SplitterAxis::Horizontal ? CLAY_SIZING_FIXED(thickness)
207 : CLAY_SIZING_GROW(0),
208 .height = context.params.axis == SplitterAxis::Vertical ? CLAY_SIZING_FIXED(thickness)
209 : CLAY_SIZING_GROW(0),
210 };
211 visual.backgroundColor =
212 context.params.backgroundColor.value_or(splitterTheme.backgroundColor);
213 if (separate_surfaces && state.isDragging)
214 visual.backgroundColor =
215 context.params.draggingColor.value_or(splitterTheme.draggingColor);
216 else if (separate_surfaces && isHovered)
217 visual.backgroundColor = context.params.hoverColor.value_or(splitterTheme.hoverColor);
218 visual.cornerRadius = context.params.cornerRadius.value_or(splitterTheme.cornerRadius);
219 visual.border = {.color = context.params.borderColor.value_or(splitterTheme.borderColor),
220 .width = context.params.borderWidth.value_or(splitterTheme.borderWidth)};
221
222 const auto surface = [&](Clay_Color color) {
223 Clay_ElementDeclaration declaration{};
224 const float extent = (hitThickness - thickness) * 0.5f;
225 declaration.layout.sizing = {
226 .width = context.params.axis == SplitterAxis::Horizontal ? CLAY_SIZING_FIXED(extent)
227 : CLAY_SIZING_GROW(0),
228 .height = context.params.axis == SplitterAxis::Vertical ? CLAY_SIZING_FIXED(extent)
229 : CLAY_SIZING_GROW(0)};
230 declaration.backgroundColor = color;
231 return declaration;
232 };
233 CLAY(rootId, root) {
234 if (separate_surfaces) {
235 CLAY(FlowUi::ValueOrThrow(context.clayID("leading-surface")),
236 surface(context.params.leadingSurfaceColor.value_or(Clay_Color{})));
237 }
238 CLAY(FlowUi::ValueOrThrow(context.clayID("visual")), visual);
239 if (separate_surfaces) {
240 CLAY(FlowUi::ValueOrThrow(context.clayID("trailing-surface")),
241 surface(context.params.trailingSurfaceColor.value_or(Clay_Color{})));
242 }
243 }
244 }

251 static void requestResizeCursor(InteractionContext& context) {
252 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get();
253 const auto& splitterTheme = theme.splitterHandleTheme;
254 const CursorType cursor =
255 context.params.axis == SplitterAxis::Horizontal
256 ? context.params.horizontalCursor.value_or(splitterTheme.horizontalResizeCursor)
257 : context.params.verticalCursor.value_or(splitterTheme.verticalResizeCursor);
258 context.uiManager.requestCursor(
259 cursor, context.params.cursorPriority.value_or(splitterTheme.cursorPriority));
260 }

401[[nodiscard]] inline std::span<const DevClayNode> fullClaySubtree(
402 const DevTreeSnapshot& snapshot, DevFlowNodeIndex flow) noexcept {
403 if (flow >= snapshot.flow.nodes.size()) return {};
404 const DevClayNodeIndex root = snapshot.flow.nodes[flow].clayRoot;
405 if (root >= snapshot.clay.nodes.size()) return {};
406 const uint32_t end = snapshot.clay.nodes[root].subtreeEnd;
407 if (end < root || end > snapshot.clay.nodes.size()) return {};
408 return std::span<const DevClayNode>(snapshot.clay.nodes).subspan(root, end - root);
409}

412[[nodiscard]] inline std::span<const DevClayNodeIndex> directClayContribution(
413 const DevTreeSnapshot& snapshot, DevFlowNodeIndex flow) noexcept {
414 if (flow >= snapshot.flow.nodes.size()) return {};
415 const DevFlowNode& node = snapshot.flow.nodes[flow];
416 if (node.firstDirectClay > snapshot.flow.directClayNodes.size() ||
417 node.directClayCount > snapshot.flow.directClayNodes.size() - node.firstDirectClay) return {};
418 return std::span<const DevClayNodeIndex>(snapshot.flow.directClayNodes)
419 .subspan(node.firstDirectClay, node.directClayCount);
420}

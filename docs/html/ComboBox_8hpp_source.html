206 static void onHovered(InteractionContext& context) {
207 if (!isEnabled(context.params)) return;
208 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().comboBoxTheme;
209 context.uiManager.requestCursor(
210 context.params.cursor.value_or(theme.cursor),
211 context.params.cursorPriority.value_or(theme.cursorPriority));
212 }
213
218 static void onPressed(InteractionContext& context) {
219 if (!isEnabled(context.params)) {
220 clearTrigger(context.state());
221 return;
222 }
223 context.state().triggerArmed = true;
224 context.state().triggerMouseUpObserved = false;
225 }
226
231 static void onReleased(InteractionContext& context) {
232 if (!context.state().triggerArmed) return;
233 clearTrigger(context.state());
234 if (isEnabled(context.params)) {
235 setOpen(context, !resolvedOpen(context));
236 }
237 }
238
243 static void runLogic(InteractionContext& context) {
244 auto& state = context.state();
245 if (context.params.open) state.open = *context.params.open;
246
247 if (context.uiManager.popups().consumeDismissed(FlowUi::ValueOrThrow(context.part(Parts::popup)))) {
248 setOpen(context, false);
249 }
250
251 if (!isEnabled(context.params)) {
252 if (resolvedOpen(context)) setOpen(context, false);
253 clearTrigger(state);
254 clearOption(state);
255 return;
256 }
257
258 const InteractionSnapshot& interaction =
259 context.uiManager.getPreviousFramesInteraction();
260 for (const ComboBoxOption& option : context.params.options) {
261 if (!option.enabled) continue;
262 const Clay_ElementId optionId = FlowUi::ValueOrThrow(context.clayID(Keyed("option", option.value)));
263 if (interaction.isPressed(optionId)) {
264 state.optionArmed = true;
265 state.optionMouseUpObserved = false;
266 state.armedOptionValue = option.value;
267 }
268 if (state.optionArmed && state.armedOptionValue == option.value &&
269 interaction.isReleased(optionId)) {
270 const bool changed = *context.params.selectedValue != option.value;
271 *context.params.selectedValue = option.value;
272 clearOption(state);
273 if (changed) (void)context.invoke(context.params.onChanged);
274 setOpen(context, false);
275 break;
276 }
277 }
278
279 const bool pointerDown = context.uiManager.getCurrentFrameInput().mouseDown[0];
280 maintainArm(pointerDown, state.triggerArmed, state.triggerMouseUpObserved);
281 maintainArm(pointerDown, state.optionArmed, state.optionMouseUpObserved);
282 }

288 static void buildElement(BuildContext& context) {
289 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().comboBoxTheme;
290 const bool enabled = isEnabled(context.params);
291 const bool open = context.params.open ? *context.params.open : context.state().open;
292 const ComboBoxOption* selected = selectedOption(context.params);
293 const FSELComboBoxStateTheme appearance = resolveTriggerAppearance(context, enabled, open);
294
295 Clay_ElementDeclaration trigger{};
296 trigger.layout.sizing = context.params.sizing.value_or(Clay_Sizing{
297 .width = CLAY_SIZING_FIXED(theme.width),
298 .height = CLAY_SIZING_FIXED(theme.height),
299 });
300 const auto retained = Clay_GetElementData(context.clayID());
301 const auto content_layout = detail::combo_box::fit_content(
302 detail::combo_box::resolved_extent(trigger.layout.sizing.width,
303 retained.found ? retained.boundingBox.width : 0.0f,
304 theme.width),
305 detail::combo_box::resolved_extent(trigger.layout.sizing.height,
306 retained.found ? retained.boundingBox.height : 0.0f,
307 theme.height),
308 theme.triggerPadding, context.params.iconSize.value_or(theme.iconSize),
309 theme.contentGap, selected && selected->icon.handle, true,
310 context.params.fontSize.value_or(theme.fontSize));
311 trigger.layout.padding = content_layout.padding;
312 trigger.layout.childGap = content_layout.gap;
313 trigger.layout.childAlignment = {
314 .x = CLAY_ALIGN_X_LEFT,
315 .y = CLAY_ALIGN_Y_CENTER,
316 };
317 trigger.layout.layoutDirection = CLAY_LEFT_TO_RIGHT;
318 trigger.backgroundColor = appearance.backgroundColor;
319 trigger.cornerRadius = theme.cornerRadius;
320 trigger.border = {.color = appearance.borderColor, .width = theme.borderWidth};
321 trigger.clip = {
322 .horizontal = true,
323 .vertical = true,
324 .scrollInputDisabled = true,
325 };
326
327 Clay_TextElementConfig textConfig =
328 makeTextConfig(context, selected ? appearance.textColor : theme.placeholderColor);
329 textConfig.fontSize = static_cast<uint16_t>(
330 std::max(1.0f, std::min<float>(textConfig.fontSize, content_layout.content_height)));
331 textConfig.textAlignment = context.params.labelAlignment;
332 CLAY(context.clayID(), trigger) {
333 if (selected && content_layout.show_leading_icon) {
334 drawIcon(context, FlowUi::ValueOrThrow(context.clayID("selected-icon")), selected->icon,
335 appearance.iconColor, content_layout.icon_size);
336 }
337 Clay_ElementDeclaration labelRoot{};
338 labelRoot.layout.sizing = {
339 .width = content_layout.label_width > 0.0f
340 ? CLAY_SIZING_GROW(0, content_layout.label_width) : CLAY_SIZING_FIXED(0),
341 .height = CLAY_SIZING_FIT(0),
342 };
343 labelRoot.layout.childAlignment.x =
344 context.params.labelAlignment == CLAY_TEXT_ALIGN_CENTER ? CLAY_ALIGN_X_CENTER
345 : context.params.labelAlignment == CLAY_TEXT_ALIGN_RIGHT ? CLAY_ALIGN_X_RIGHT
346 : CLAY_ALIGN_X_LEFT;
347 labelRoot.clip = {
348 .horizontal = true,
349 .vertical = false,
350 .scrollInputDisabled = true,
351 };
352 CLAY(FlowUi::ValueOrThrow(context.clayID("label")), labelRoot) {
353 CLAY_TEXT(context.uiManager.toClayString(selected ? selected->text
354 : context.params.placeholder),
355 CLAY_TEXT_CONFIG(textConfig));
356 }
357 drawDisclosure(context, open, appearance.iconColor, content_layout.icon_size);
358 }
359
360 if (open)
361 drawPopup(context, theme);
362 }

369 static const ComboBoxOption* selectedOption(const Parameters& params) {
370 if (!params.selectedValue) return nullptr;
371 for (const ComboBoxOption& option : params.options) {
372 if (option.value == *params.selectedValue) return &option;
373 }
374 return nullptr;

377 static bool resolvedOpen(const InteractionContext& context) {
378 return context.params.open ? *context.params.open : context.state().open;

381 static void setOpen(InteractionContext& context, bool value) {
382 const bool old = resolvedOpen(context);
383 context.state().open = value;
384 if (context.params.open) *context.params.open = value;
385 if (old == value) return;
386 (void)context.invoke(value ? context.params.onOpened : context.params.onClosed);
387 }

395 state.optionArmed = false;

400 static void maintainArm(bool pointerDown, bool& armed, bool& mouseUpObserved) {
401 if (!armed) return;
402 if (pointerDown) {
403 if (mouseUpObserved) {
404 armed = false;
405 mouseUpObserved = false;
406 }
407 return;
408 }
409 if (mouseUpObserved) {
410 armed = false;
411 mouseUpObserved = false;
412 } else {
413 mouseUpObserved = true;
414 }

490 static void drawPopup(BuildContext& context, const FSELComboBoxTheme& theme) {
491 const float optionHeight =
492 std::max(context.params.optionHeight.value_or(theme.optionHeight), 1.0f);
493 const auto& surface = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().popupSurfaceTheme;
494 const float minimumPopupHeight =
495 optionHeight + static_cast<float>(surface.padding.top + surface.padding.bottom);
496 const float maximumHeight = std::max(
497 context.params.popupMaxHeight.value_or(theme.popupMaxHeight), minimumPopupHeight);
498 const float contentHeight =
499 context.params.options.empty()
500 ? optionHeight
501 : optionHeight * static_cast<float>(context.params.options.size()) +
502 static_cast<float>(theme.optionGap) *
503 static_cast<float>(context.params.options.size() - 1u);
504 const float searchHeight = context.params.popupSearchValue
505 ? std::max(context.params.popupSearchHeight.value_or(30.0f), 1.0f) : 0.0f;
506 const float searchGap = context.params.popupSearchValue
507 ? static_cast<float>(theme.scrollGap) : 0.0f;
508 const float popupHeight = std::min(
509 contentHeight + static_cast<float>(surface.padding.top + surface.padding.bottom),
510 maximumHeight) + searchHeight + searchGap;
511
512 const Clay_ElementData triggerData = Clay_GetElementData(context.clayID());
513 const float triggerWidth = triggerData.found && triggerData.boundingBox.width > 0.0f
514 ? triggerData.boundingBox.width : theme.width;
515 const float requestedWidth = std::max(
516 context.params.popupWidth.value_or(theme.width), 1.0f);
517 float expectedWidth = triggerWidth;
518 Clay_SizingAxis popupWidthSizing = CLAY_SIZING_FIXED(triggerWidth);
519 switch (context.params.popupWidthPolicy) {
520 case ComboBoxPopupWidthPolicy::MatchTrigger:
521 break;
522 case ComboBoxPopupWidthPolicy::Fixed:
523 expectedWidth = requestedWidth;
524 popupWidthSizing = CLAY_SIZING_FIXED(requestedWidth);
525 break;
526 case ComboBoxPopupWidthPolicy::ContentAtLeastTrigger:
527 expectedWidth = 0.0f;
528 popupWidthSizing = CLAY_SIZING_FIT(
529 triggerWidth,
530 std::max(triggerWidth, requestedWidth));
531 break;
532 }
533
534 PopupRequest request{};
535 request.anchor = PopupAnchor::element(context.id);
536 request.placement = context.params.placement;
537 request.overflow = context.params.overflow;
538 request.layer = context.params.layer;
539 request.pointerCaptureMode = context.params.pointerCaptureMode;
540 request.outsidePress = context.params.outsidePress;
541 if (expectedWidth > 0.0f) {
542 request.expectedSize = Clay_Dimensions{expectedWidth, popupHeight};
543 }
544 const FlowElementPartID popupId = FlowUi::ValueOrThrow(context.part(Parts::popup));
545 const PopupFrame frame = context.uiManager.popups().request(popupId, request).value_or(PopupFrame{});
546 if (!frame.visible && !frame.measureOnly) return;
547
548 Clay_ElementDeclaration popup{};
549 popup.layout.sizing = {
550 .width = popupWidthSizing,
551 .height = CLAY_SIZING_FIXED(popupHeight),
552 };
553 popup.layout.padding = surface.padding;
554 popup.layout.childGap = context.params.popupSearchValue ? theme.scrollGap : 0;
555 popup.layout.childAlignment = {.x = CLAY_ALIGN_X_LEFT, .y = CLAY_ALIGN_Y_TOP};
556 popup.layout.layoutDirection = CLAY_TOP_TO_BOTTOM;
557 popup.backgroundColor = surface.backgroundColor;
558 popup.cornerRadius = surface.cornerRadius;
559 popup.border = {.color = surface.borderColor, .width = surface.borderWidth};
560 popup.floating = frame.floating;
561
562 const Clay_ElementId scrollId = FlowUi::ValueOrThrow(context.clayID("scroll-viewport"));
563 const Clay_ScrollContainerData scroll = Clay_GetScrollContainerData(scrollId);
564 const float trackHeight = std::max(
565 popupHeight - searchHeight - searchGap -
566 static_cast<float>(surface.padding.top + surface.padding.bottom),
567 0.0f);
568 const auto thumb = scroll.found && scroll.scrollPosition
569 ? detail::scroll_indicator::calculate(
570 scroll.scrollContainerDimensions.height,
571 scroll.contentDimensions.height,
572 scroll.scrollPosition->y,
573 trackHeight,
574 theme.scrollThumbMinimum)
575 : detail::scroll_indicator::ThumbGeometry{};
576
577 CLAY(context.uiManager.toClayEID(popupId), popup) {
578 if (context.params.popupSearchValue) {
579 Clay_ElementDeclaration searchSlot{};
580 searchSlot.layout.sizing = {
581 .width = CLAY_SIZING_GROW(0), .height = CLAY_SIZING_FIXED(searchHeight)};
582 searchSlot.layout.childAlignment = {
583 .x = CLAY_ALIGN_X_LEFT, .y = CLAY_ALIGN_Y_CENTER};
584 searchSlot.clip = {.horizontal = true, .vertical = true};
585 CLAY(FlowUi::ValueOrThrow(context.clayID("popup-search-slot")), searchSlot) {
586 TextInputParameters search{};
587 search.value = context.params.popupSearchValue;
588 search.syncPolicy = TextFieldSyncPolicy::Live;
589 search.placeholder = context.params.popupSearchPlaceholder;
590 search.enabled = context.params.enabled;
591 search.readOnly = !context.params.enabled;
592 search.maxBytes = 96u;
593 search.sizing = Clay_Sizing{
594 .width = CLAY_SIZING_GROW(0), .height = CLAY_SIZING_GROW(0)};
595 // Match the working 30 px inspector search fields: the default 8 px
596 // vertical padding leaves too little height for a managed text line.
597 search.padding = Clay_Padding{8, 8, 5, 5};
598 search.fontSize = context.params.fontSize.value_or(theme.fontSize);
599 search.floatingZIndex = frame.floating.zIndex;
600 context.uiManager.createElement(kTextInput, "popup-search")
601 .setParameters(std::move(search)).setDevInternalCapture(true).draw();
602 }
603 }
604 Clay_ElementDeclaration contentRow{};
605 contentRow.layout.sizing = {
606 .width = CLAY_SIZING_GROW(0), .height = CLAY_SIZING_GROW(0)};
607 contentRow.layout.layoutDirection = CLAY_LEFT_TO_RIGHT;
608 contentRow.layout.childGap = theme.scrollGap;
609 CLAY(FlowUi::ValueOrThrow(context.clayID("popup-content")), contentRow) {
610 Clay_ElementDeclaration viewport{};
611 viewport.layout.sizing = {
612 .width = CLAY_SIZING_GROW(0),
613 .height = CLAY_SIZING_GROW(0),
614 };
615 viewport.layout.layoutDirection = CLAY_TOP_TO_BOTTOM;
616 viewport.clip = {
617 .vertical = true,
618 .childOffset = scroll.found && scroll.scrollPosition
619 ? *scroll.scrollPosition : Clay_Vector2{},
620 };
621 CLAY(scrollId, viewport) {
622 Clay_ElementDeclaration options{};
623 options.layout.sizing = {
624 .width = CLAY_SIZING_GROW(0),
625 .height = CLAY_SIZING_FIT(0),
626 };
627 options.layout.childGap = theme.optionGap;
628 options.layout.layoutDirection = CLAY_TOP_TO_BOTTOM;
629 CLAY(FlowUi::ValueOrThrow(context.clayID("options")), options) {
630 for (const ComboBoxOption& option : context.params.options) {
631 drawOption(context, option, optionHeight, theme);
632 }
633 }
634 }
635 if (context.params.showScrollIndicator && thumb.visible) {
636 drawScrollIndicator(context, thumb, trackHeight, theme);
637 }
638 }
639 }
640 }

642 static void drawOption(
643 BuildContext& context,
644 const ComboBoxOption& option,
645 float height,
646 const FSELComboBoxTheme& theme) {
647 const Clay_ElementId id = FlowUi::ValueOrThrow(context.clayID(Keyed("option", option.value)));
648 const InteractionSnapshot& interaction =
649 context.uiManager.getPreviousFramesInteraction();
650 FSELComboBoxOptionStateTheme appearance = theme.optionIdle;
651 if (!option.enabled) {
652 appearance = theme.optionDisabled;
653 } else if (context.state().optionArmed &&
654 context.state().armedOptionValue == option.value &&
655 context.uiManager.getCurrentFrameInput().mouseDown[0]) {
656 appearance = theme.optionPressed;
657 } else if (interaction.isHovered(id)) {
658 appearance = theme.optionHovered;
659 } else if (context.params.selectedValue && *context.params.selectedValue == option.value) {
660 appearance = theme.optionSelected;
661 }
662
663 Clay_ElementDeclaration row{};
664 row.layout.sizing = {
665 .width = CLAY_SIZING_GROW(0),
666 .height = CLAY_SIZING_FIXED(height),
667 };
668 const auto retained = Clay_GetElementData(id);
669 const auto content_layout = detail::combo_box::fit_content(
670 retained.found && retained.boundingBox.width > 0.0f
671 ? retained.boundingBox.width
672 : context.params.popupWidth.value_or(theme.width),
673 height, theme.optionPadding, context.params.iconSize.value_or(theme.iconSize),
674 theme.contentGap, static_cast<bool>(option.icon.handle), false,
675 context.params.fontSize.value_or(theme.fontSize));
676 row.layout.padding = content_layout.padding;
677 row.layout.childGap = content_layout.gap;
678 row.layout.childAlignment = {.x = CLAY_ALIGN_X_LEFT, .y = CLAY_ALIGN_Y_CENTER};
679 row.layout.layoutDirection = CLAY_LEFT_TO_RIGHT;
680 row.backgroundColor = appearance.backgroundColor;
681 row.cornerRadius = theme.cornerRadius;
682 row.clip = {.horizontal = true, .vertical = true, .scrollInputDisabled = true};
683 CLAY(id, row) {
684 if (content_layout.show_leading_icon) {
685 drawIcon(context, FlowUi::ValueOrThrow(context.clayID(Keyed("option-icon", option.value))), option.icon,
686 appearance.iconColor, content_layout.icon_size);
687 }
688 Clay_ElementDeclaration labelSlot{};
689 labelSlot.layout.sizing = {
690 .width = content_layout.label_width > 0.0f
691 ? CLAY_SIZING_GROW(0, content_layout.label_width) : CLAY_SIZING_FIXED(0),
692 .height = CLAY_SIZING_FIT(0)};
693 labelSlot.clip = {.horizontal = true, .vertical = true, .scrollInputDisabled = true};
694 CLAY(FlowUi::ValueOrThrow(context.clayID(Keyed("option-label", option.value))), labelSlot) {
695 auto text_config = makeTextConfig(context, appearance.textColor);
696 text_config.fontSize = static_cast<uint16_t>(std::max(
697 1.0f, std::min<float>(text_config.fontSize, content_layout.content_height)));
698 CLAY_TEXT(context.uiManager.toClayString(option.text),
699 CLAY_TEXT_CONFIG(text_config));
700 }
701 }
702 }

704 static void drawScrollIndicator(BuildContext& context,
705 const detail::scroll_indicator::ThumbGeometry& thumb,
706 float trackHeight, const FSELComboBoxTheme& theme) {
707 Clay_ElementDeclaration track{};

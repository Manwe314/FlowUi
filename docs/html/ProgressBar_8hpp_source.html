73 static void buildElement(BuildContext& context) {
74 const FSELProgressBarTheme& theme =
75 FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().progressBarTheme;
76 const detail::bounded_value::Bounds bounds =
77 detail::bounded_value::normalizeBounds(
78 context.params.minimum,
79 context.params.maximum);
80 const float ratio = static_cast<float>(
81 detail::bounded_value::normalizedRatio(
82 context.params.value,
83 bounds));
84 const Clay_BorderWidth borderWidth =
85 context.params.borderWidth.value_or(theme.borderWidth);
86 const Clay_CornerRadius cornerRadius =
87 context.params.cornerRadius.value_or(theme.cornerRadius);
88
89 Clay_ElementDeclaration root{};
90 root.layout.sizing = context.params.sizing.value_or(
91 defaultSizing(context.params.axis, theme));
92 root.layout.padding = {
93 .left = borderWidth.left,
94 .right = borderWidth.right,
95 .top = borderWidth.top,
96 .bottom = borderWidth.bottom,
97 };
98 root.layout.childAlignment = childAlignment(
99 context.params.axis,
100 context.params.inverted);
101 root.layout.layoutDirection =
102 context.params.axis == ProgressBarAxis::Horizontal
103 ? CLAY_LEFT_TO_RIGHT
104 : CLAY_TOP_TO_BOTTOM;
105 root.backgroundColor = context.params.baseColor.value_or(
106 theme.baseColor);
107 root.cornerRadius = cornerRadius;
108 root.clip = {
109 .horizontal = true,
110 .vertical = true,
111 };
112 root.border = {
113 .color = context.params.borderColor.value_or(theme.borderColor),
114 .width = borderWidth,
115 };
116
117 Clay_ElementDeclaration fill{};
118 fill.layout.sizing = fillSizing(context.params.axis, ratio);
119 fill.backgroundColor = context.params.fillColor.value_or(
120 theme.fillColor);
121 fill.cornerRadius = cornerRadius;
122
123 CLAY(context.clayID(), root) {
124 if (ratio > 0.0f) {
125 CLAY(FlowUi::ValueOrThrow(context.clayID("fill")), fill);
126 }
127 }
128 }

131 static Clay_Sizing defaultSizing(
132 ProgressBarAxis axis,
133 const FSELProgressBarTheme& theme) {
134 const float length = std::max(theme.length, 1.0f);
135 const float thickness = std::max(theme.thickness, 1.0f);
136 return {
137 .width = axis == ProgressBarAxis::Horizontal
138 ? CLAY_SIZING_FIXED(length)
139 : CLAY_SIZING_FIXED(thickness),
140 .height = axis == ProgressBarAxis::Vertical
141 ? CLAY_SIZING_FIXED(length)
142 : CLAY_SIZING_FIXED(thickness),
143 };
144 }

146 static Clay_ChildAlignment childAlignment(
147 ProgressBarAxis axis,
148 bool inverted) {
149 if (axis == ProgressBarAxis::Horizontal) {
150 return {
151 .x = inverted ? CLAY_ALIGN_X_RIGHT : CLAY_ALIGN_X_LEFT,
152 .y = CLAY_ALIGN_Y_CENTER,
153 };
154 }
155 return {
156 .x = CLAY_ALIGN_X_CENTER,
157 .y = inverted ? CLAY_ALIGN_Y_TOP : CLAY_ALIGN_Y_BOTTOM,
158 };
159 }
160
161 static Clay_Sizing fillSizing(ProgressBarAxis axis, float ratio) {
162 if (axis == ProgressBarAxis::Horizontal) {
163 return {
164 .width = CLAY_SIZING_PERCENT(ratio),
165 .height = CLAY_SIZING_GROW(0),
166 };
167 }
168 return {
169 .width = CLAY_SIZING_GROW(0),
170 .height = CLAY_SIZING_PERCENT(ratio),
171 };
172 }

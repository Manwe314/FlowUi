138 static void buildElement(BuildContext& context) {
139 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().switchTheme;
140 const FSELSwitchStateTheme appearance = resolveAppearance(context, theme);
141 const float trackWidth = std::max(
142 context.params.trackWidth.value_or(theme.trackWidth),
143 1.0f);
144 const float trackHeight = std::max(
145 context.params.trackHeight.value_or(theme.trackHeight),
146 1.0f);
147 const float maximumInset = std::max(
148 (std::min(trackWidth, trackHeight) - 1.0f) * 0.5f,
149 0.0f);
150 const uint16_t knobInset = static_cast<uint16_t>(std::min(
151 static_cast<float>(context.params.knobInset.value_or(theme.knobInset)),
152 maximumInset));
153 const float knobSize = std::max(
154 std::min(trackWidth, trackHeight) - 2.0f * knobInset,
155 1.0f);
156 const float roundness = detail::boolean_control::clampRoundness(
157 context.params.roundness.value_or(theme.roundness));
158
159 Clay_ElementDeclaration track{};
160 track.layout.sizing = {
161 .width = CLAY_SIZING_FIXED(trackWidth),
162 .height = CLAY_SIZING_FIXED(trackHeight),
163 };
164 track.layout.padding = {
165 .left = knobInset,
166 .right = knobInset,
167 .top = knobInset,
168 .bottom = knobInset,
169 };
170 track.layout.childAlignment = {
171 .x = context.params.isOn ? CLAY_ALIGN_X_RIGHT : CLAY_ALIGN_X_LEFT,
172 .y = CLAY_ALIGN_Y_CENTER,
173 };
174 track.backgroundColor = appearance.trackColor;
175 track.cornerRadius = CLAY_CORNER_RADIUS(
176 std::min(trackWidth, trackHeight) * 0.5f * roundness);
177 track.border = {
178 .color = appearance.trackBorderColor,
179 .width = context.params.trackBorderWidth.value_or(
180 theme.trackBorderWidth),
181 };
182
183 Clay_ElementDeclaration knob{};
184 knob.layout.sizing = {
185 .width = CLAY_SIZING_FIXED(knobSize),
186 .height = CLAY_SIZING_FIXED(knobSize),
187 };
188 knob.backgroundColor = appearance.knobColor;
189 knob.cornerRadius = CLAY_CORNER_RADIUS(
190 knobSize * 0.5f * roundness);
191 knob.border = {
192 .color = appearance.knobBorderColor,
193 .width = context.params.knobBorderWidth.value_or(
194 theme.knobBorderWidth),
195 };
196
197 CLAY(context.clayID(), track) {
198 CLAY(FlowUi::ValueOrThrow(context.clayID("knob")), knob);
199 }

217 static FSELSwitchStateTheme resolveAppearance(
218 BuildContext& context,
219 const FSELSwitchTheme& theme) {
220 const FSELSwitchValueTheme& valueTheme = context.params.isOn
221 ? theme.on
222 : theme.off;
223 const SwitchValueOverrides& overrides = context.params.isOn
224 ? context.params.onOverrides
225 : context.params.offOverrides;
226
227 switch (detail::boolean_control::resolveVisualState(
228 context,
229 context.params.onToggle)) {
230 case detail::boolean_control::VisualState::Idle:
231 return applyOverrides(valueTheme.idle, overrides.idle);
232 case detail::boolean_control::VisualState::Hovered:
233 return applyOverrides(valueTheme.hovered, overrides.hovered);
234 case detail::boolean_control::VisualState::Pressed:
235 return applyOverrides(valueTheme.pressed, overrides.pressed);
236 case detail::boolean_control::VisualState::Disabled:
237 return applyOverrides(valueTheme.disabled, overrides.disabled);
238 }
239 return applyOverrides(valueTheme.idle, overrides.idle);
240 }

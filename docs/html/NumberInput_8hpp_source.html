309 static void runLogic(InteractionContext& context) {
310 if (!hasStepButtons(context.params)) {
311 return;
312 }
313 const bool increment = context.previousInteraction.isPressed(
314 FlowUi::ValueOrThrow(context.part(Parts::increment)));
315 const bool decrement = context.previousInteraction.isPressed(
316 FlowUi::ValueOrThrow(context.part(Parts::decrement)));
317 if (increment == decrement) {
318 return;
319 }
320 applyStep(context, increment ? 1 : -1);

327 static void buildElement(BuildContext& context) {
328 const FSELNumberInputTheme& numberTheme =
329 FlowUi::ValueOrThrow(context.uiManager.template theme<FSELTheme>()).get().numberInputTheme;
330 const FSELTextFieldTheme& theme = numberTheme.field;
331 auto& state = context.state();
332 auto& manager = context.uiManager.inputFields();
333 const detail::numeric::Bounds<T> bounds = detail::numeric::resolveBounds(
334 context.params.minimum,
335 context.params.maximum);
336 const bool enabled = isBoundAndEnabled(context.params) && bounds.valid;
337 const bool editable = enabled && !context.params.readOnly;
338 const bool buttonsVisible = hasStepButtons(context.params);
339 if (!enabled && state.wasFocused) {
340 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
341 }
342
343 const Clay_ElementId contentId = context.uiManager.toClayEID(
344 FlowUi::ValueOrThrow(context.part(Parts::content)));
345 const Clay_ElementId textId = context.uiManager.toClayEID(
346 FlowUi::ValueOrThrow(context.part(Parts::text)));
347 const Clay_ElementId placeholderId = context.uiManager.toClayEID(
348 FlowUi::ValueOrThrow(context.part(Parts::placeholder)));
349 const Clay_ElementId decrementId = context.uiManager.toClayEID(
350 FlowUi::ValueOrThrow(context.part(Parts::decrement)));
351 const Clay_ElementId incrementId = context.uiManager.toClayEID(
352 FlowUi::ValueOrThrow(context.part(Parts::increment)));
353 const std::array<Clay_ElementId, 2> focusRetentionIds{
354 decrementId,
355 incrementId,
356 };
357
358 const Clay_Padding padding = context.params.padding.value_or(theme.padding);
359 const Clay_Dimensions viewport = detail::text_field::resolveViewport(
360 contentId,
361 theme,
362 padding,
363 context.params.viewportWidth,
364 context.params.viewportHeight);
365 Clay_TextElementConfig textConfig{};
366 textConfig.fontId = context.uiManager.resolveFont(
367 context.params.fontFamily.value_or(theme.fontFamily),
368 context.params.fontWeight.value_or(theme.fontWeight),
369 context.params.fontStyle.value_or(theme.fontStyle));
370 textConfig.fontSize = context.params.fontSize.value_or(theme.fontSize);
371 textConfig.letterSpacing = context.params.letterSpacing.value_or(
372 theme.letterSpacing);
373 textConfig.wrapMode = CLAY_TEXT_WRAP_NONE;
374 textConfig.textAlignment = CLAY_TEXT_ALIGN_LEFT;
375
376 const T currentValue = context.params.value ? *context.params.value : T{};
377 const detail::numeric_text::FormattedValue initialText =
378 detail::numeric_text::format(currentValue, context.params.format);
379 FieldRequest request{
380 .initialText = initialText.view(),
381 .config = FieldConfig{
382 .maxBytes = context.params.maxBytes,
383 .mode = TextFieldMode::SingleLine,
384 .transactionDetail = context.params.transactionDetail,
385 .readOnly = !editable,
386 .softWrap = false,
387 .allowArrowNavigation = true,
388 },
389 .layout = TextLayoutDescriptor{
390 .fontId = textConfig.fontId,
391 .fontSize = textConfig.fontSize,
392 .letterSpacing = textConfig.letterSpacing,
393 .viewportWidth = viewport.width,
394 .viewportHeight = viewport.height,
395 .tabWidth = theme.tabWidth,
396 .floatingZIndex = context.params.floatingZIndex.value_or(0),
397 },
398 .overlayStyle = detail::text_field::resolveOverlayStyle(
399 theme,
400 context.params.caret),
401 .textElementId = enabled ? textId : Clay_ElementId{},
402 .contentElementId = enabled ? contentId : Clay_ElementId{},
403 .focusRetentionElementIds = buttonsVisible && enabled
404 ? std::span<const Clay_ElementId>(focusRetentionIds)
405 : std::span<const Clay_ElementId>{},
406 };
407 FieldQueryResult field = manager.requestField(context.id, request);
408
409 const bool userTextChanged = std::ranges::any_of(
410 field.transactions,
411 [](const FieldEditTransaction& transaction) {
412 return transaction.origin != EditOrigin::Programmatic;
413 });
414 if (editable && userTextChanged) {
415 const std::string fieldText = FlowUi::ValueOrThrow(field.text.copy());
416 const std::string sanitized = detail::numeric_text::sanitize<T>(
417 fieldText,
418 context.params.format.allowScientificInput);
419 if (sanitized != fieldText) {
420 (void)manager.replaceText(context.id, sanitized, true);
421 field = manager.requestField(context.id, request);
422 }
423 }
424
425 bool focused = enabled && field.hasPrimaryCaret;
426 if (state.forceTextSynchronization ||
427 (!focused && !userTextChanged && !state.editSession.active &&
428 !detail::text_field::textEquals(field.text, initialText.view()))) {
429 (void)manager.replaceText(context.id, initialText.view(), focused);
430 field = manager.requestField(context.id, request);
431 focused = enabled && field.hasPrimaryCaret;
432 state.forceTextSynchronization = false;
433 }
434
435 const detail::numeric_text::ClassifiedValue<T> classified =
436 detail::numeric_text::classify(
437 detail::numeric_text::parse<T>(FlowUi::ValueOrThrow(field.text.copy())),
438 bounds);
439 const std::optional<T> candidate = classified.candidate;
440 state.textStatus = classified.status;
441
442 if (!bounds.valid || !detail::numeric::isFinite(currentValue)) {
443 state.textStatus = NumberInputTextStatus::ConfigurationError;
444 }
445
446 if (userTextChanged && editable) {
447 detail::numeric_control::beginEdit<T>(context);
448 if (candidate.has_value()) {
449 state.editSession.hasLastValidValue = true;
450 state.editSession.lastValidValue = *candidate;
451 if (context.params.syncPolicy == NumberInputSyncPolicy::Live) {
452 (void)detail::numeric_control::applyValue<T>(
453 context,
454 *candidate);
455 }
456 }
457 }
458
459 const bool cancelled = detail::text_field::hasCommand(
460 field,
461 FieldCommandRequest::Cancel);
462 const bool submitted = detail::text_field::hasCommand(
463 field,
464 FieldCommandRequest::Submit);
465 const bool blurred = state.wasFocused && !focused;
466 if (!state.wasFocused && focused) {
467 invoke(context, context.params.onFocus);
468 }
469
470 if (cancelled) {
471 detail::numeric_control::cancelEdit<T>(context);
472 const detail::numeric_text::FormattedValue restored =
473 detail::numeric_text::format(
474 context.params.value ? *context.params.value : T{},
475 context.params.format);
476 (void)manager.replaceText(context.id, restored.view(), true);
477 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
478 field = manager.requestField(context.id, request);
479 focused = false;
480 } else if (submitted || blurred) {
481 detail::numeric_control::commitTextEdit<T>(
482 context,
483 candidate,
484 context.params.syncPolicy);
485 const detail::numeric_text::FormattedValue committed =
486 detail::numeric_text::format(
487 context.params.value ? *context.params.value : T{},
488 context.params.format);
489 (void)manager.replaceText(context.id, committed.view(), focused);
490 field = manager.requestField(context.id, request);
491 if (submitted) {
492 invoke(context, context.params.onSubmit);
493 if (context.params.clearFocusOnSubmit) {
494 manager.requestCaret(context.id, CaretRequestKind::ClearAll);
495 focused = false;
496 }
497 }
498 }
499
500 if (submitted && context.params.clearFocusOnSubmit) {
501 invoke(context, context.params.onBlur);
502 } else if (blurred || cancelled) {
503 invoke(context, context.params.onBlur);
504 }
505 if (detail::text_field::hasCommand(field, FieldCommandRequest::Undo)) {
506 invoke(context, context.params.onUndoRequested);
507 }
508 if (detail::text_field::hasCommand(field, FieldCommandRequest::Redo)) {
509 invoke(context, context.params.onRedoRequested);
510 }
511
512 if (context.params.boundsPolicy ==
513 NumberInputBoundsPolicy::ClampTextImmediately &&
514 candidate.has_value() &&
515 state.textStatus != NumberInputTextStatus::Complete) {
516 const detail::numeric_text::FormattedValue clamped =
517 detail::numeric_text::format(*candidate, context.params.format);
518 (void)manager.replaceText(context.id, clamped.view(), focused);
519 field = manager.requestField(context.id, request);
520 state.textStatus = NumberInputTextStatus::Complete;
521 }
522 if (submitted || blurred || cancelled) {
523 const detail::numeric_text::FormattedValue canonical =
524 detail::numeric_text::format(
525 context.params.value ? *context.params.value : T{},
526 context.params.format);
527 state.textStatus = detail::numeric_text::classify(
528 detail::numeric_text::parse<T>(canonical.view()),
529 bounds).status;
530 if (!bounds.valid || (context.params.value &&
531 !detail::numeric::isFinite(*context.params.value))) {
532 state.textStatus = NumberInputTextStatus::ConfigurationError;
533 }
534 }
535
536 state.wasFocused = focused;
537 drawField(
538 context,
539 field,
540 textConfig,
541 padding,
542 contentId,
543 textId,
544 placeholderId,
545 decrementId,
546 incrementId,
547 enabled,
548 focused,
549 numberTheme);
550 }

569 static void applyStep(InteractionContext& context, int direction) {
570 const detail::numeric::Bounds<T> bounds = detail::numeric::resolveBounds(
571 context.params.minimum,
572 context.params.maximum);
573 if (!isBoundAndEnabled(context.params) || context.params.readOnly ||
574 !bounds.valid || !detail::numeric::validStep(context.params.step) ||
575 !detail::numeric::isFinite(*context.params.value)) {
576 return;
577 }
578 const T next = detail::numeric::stepValue(
579 *context.params.value,
580 context.params.step,
581 direction,
582 bounds);
583 if (next == *context.params.value) {
584 return;
585 }
586 detail::numeric_control::beginEdit<T>(context);
587 (void)detail::numeric_control::applyValue<T>(context, next);
588 context.state().editSession.hasLastValidValue = true;
589 context.state().editSession.lastValidValue = next;
590 context.state().forceTextSynchronization = true;
591 if (!context.state().wasFocused) {
592 detail::numeric_control::commitEdit(context);
593 }
594 }

596 static FSELTextFieldStateTheme resolveAppearance(
597 BuildContext& context,
598 const FSELTextFieldTheme& theme,
599 bool enabled,
600 bool focused) {
601 if (!enabled) {
602 return detail::text_field::applyOverrides(
603 theme.disabled,
604 context.params.disabledOverrides);
605 }
606 if (!context.params.valid ||
607 context.state().textStatus != NumberInputTextStatus::Complete) {
608 return detail::text_field::applyOverrides(
609 theme.invalid,
610 context.params.invalidOverrides);
611 }
612 if (focused) {
613 return detail::text_field::applyOverrides(
614 theme.focused,
615 context.params.focusedOverrides);
616 }
617 if (context.params.readOnly) {
618 return detail::text_field::applyOverrides(
619 theme.readOnly,
620 context.params.readOnlyOverrides);
621 }
622 if (context.uiManager.getPreviousFramesInteraction().isHovered(
623 context.clayID())) {
624 return detail::text_field::applyOverrides(
625 theme.hovered,
626 context.params.hoveredOverrides);
627 }
628 return detail::text_field::applyOverrides(
629 theme.idle,
630 context.params.idleOverrides);
631 }

633 static FSELNumberInputStepButtonStateTheme resolveStepAppearance(
634 BuildContext& context,
635 Clay_ElementId id,
636 bool enabled,
637 const FSELNumberInputTheme& theme) {
638 if (!enabled) {
639 return theme.stepDisabled;

641 const InteractionSnapshot& interaction =
642 context.uiManager.getPreviousFramesInteraction();
643 if (interaction.isPressed(id) || interaction.isHeld(id)) {
644 return theme.stepPressed;
645 }
646 if (interaction.isHovered(id)) {
647 return theme.stepHovered;
648 }
649 return theme.stepIdle;

652 static void drawField(
653 BuildContext& context,
654 const FieldQueryResult& field,
655 Clay_TextElementConfig textConfig,
656 Clay_Padding padding,
657 Clay_ElementId contentId,
658 Clay_ElementId textId,
659 Clay_ElementId placeholderId,
660 Clay_ElementId decrementId,
661 Clay_ElementId incrementId,
662 bool enabled,
663 bool focused,
664 const FSELNumberInputTheme& numberTheme) {
665 const FSELTextFieldTheme& theme = numberTheme.field;
666 const FSELTextFieldStateTheme appearance = resolveAppearance(
667 context,
668 theme,
669 enabled,
670 focused);
671 textConfig.textColor = appearance.textColor;
672 Clay_TextElementConfig placeholderConfig = textConfig;
673 placeholderConfig.textColor = appearance.placeholderColor;
674
675 Clay_ElementDeclaration root{};
676 root.layout.sizing = context.params.sizing.value_or(Clay_Sizing{
677 .width = CLAY_SIZING_FIXED(std::max(theme.width, 1.0f)),
678 .height = CLAY_SIZING_FIXED(std::max(theme.height, 1.0f)),
679 });
680 root.layout.layoutDirection = CLAY_LEFT_TO_RIGHT;
681 root.backgroundColor = appearance.backgroundColor;
682 root.cornerRadius = context.params.cornerRadius.value_or(theme.cornerRadius);
683 root.clip = {.horizontal = true, .vertical = true};
684 root.border = {
685 .color = appearance.borderColor,
686 .width = context.params.borderWidth.value_or(theme.borderWidth),
687 };
688
689 Clay_ElementDeclaration content = detail::text_field::makeContentDeclaration();
690 content.layout.padding = padding;
691 const bool buttonsVisible = hasStepButtons(context.params);
692 const float buttonWidth = std::max(
693 context.params.stepButtonWidth.value_or(numberTheme.stepButtonWidth),
694 1.0f);
695 Clay_ElementDeclaration stepContainer{};
696 stepContainer.layout.sizing = {
697 .width = CLAY_SIZING_FIXED(
698 context.params.stepButtons == NumberInputStepButtons::TrailingHorizontal
699 ? buttonWidth * 2.0f
700 : buttonWidth),
701 .height = CLAY_SIZING_GROW(0),
702 };
703 stepContainer.layout.layoutDirection =
704 context.params.stepButtons == NumberInputStepButtons::TrailingHorizontal
705 ? CLAY_LEFT_TO_RIGHT
706 : CLAY_TOP_TO_BOTTOM;
707 stepContainer.border = {
708 .color = numberTheme.stepSeparatorColor,
709 .width = {.left = 1},
710 };
711
712 CLAY(context.clayID(), root) {
713 CLAY(contentId, content) {
714 if (!field.visibleLines.empty()) {
715 const VisibleTextLine& line = field.visibleLines.front();
716 auto line_declaration = line.declaration;
717 if (context.params.centerText) {
718 const auto previous_text = Clay_GetElementData(textId);
719 const auto previous_content = Clay_GetElementData(contentId);
720 const bool fits =
721 !previous_text.found || !previous_content.found ||
722 previous_text.boundingBox.width <=
723 previous_content.boundingBox.width - padding.left - padding.right;
724 line_declaration.floating.attachPoints =
725 fits ? Clay_FloatingAttachPoints{CLAY_ATTACH_POINT_CENTER_CENTER,
726 CLAY_ATTACH_POINT_CENTER_CENTER}
727 : Clay_FloatingAttachPoints{CLAY_ATTACH_POINT_LEFT_CENTER,
728 CLAY_ATTACH_POINT_LEFT_CENTER};
729 if (fits)
730 line_declaration.floating.offset.x = 0;
731 line_declaration.floating.offset.y = 0;
732 }
733 CLAY(textId, line_declaration) {
734 CLAY_TEXT(context.uiManager.toClayString(line.text),
735 CLAY_TEXT_CONFIG(textConfig));
736 }
737 }
738 if (field.text.empty() && !context.params.placeholder.empty()) {
739 CLAY(placeholderId, detail::text_field::makePlaceholderDeclaration(
740 context.params.floatingZIndex.value_or(0))) {
741 CLAY_TEXT(context.uiManager.toClayString(context.params.placeholder),
742 CLAY_TEXT_CONFIG(placeholderConfig));
743 }
744 }
745 }
746 if (buttonsVisible) {
747 CLAY(FlowUi::ValueOrThrow(context.clayID("step-buttons")), stepContainer) {
748 drawStepButton(
749 context,
750 incrementId,
751 true,
752 enabled,
753 buttonWidth,
754 numberTheme);
755 drawStepButton(
756 context,
757 decrementId,
758 false,
759 enabled,
760 buttonWidth,
761 numberTheme);
762 }
763 }
764 }
765
766 if (!field.visibleLines.empty()) {
767 const VisibleTextLine& line = field.visibleLines.front();
768 (void)context.uiManager.inputFields().submitTextSpan(
769 field.field,
770 FieldTextSpanSubmission{
771 .logicalRange = line.logicalRange,
772 .textElementId = textId,
773 .visualLineIndex = line.visualLineIndex,
774 });
775 }
776 }

778 static void drawStepButton(
779 BuildContext& context,
780 Clay_ElementId id,
781 bool increment,
782 bool fieldEnabled,
783 float buttonWidth,
784 const FSELNumberInputTheme& theme) {
785 const bool enabled = fieldEnabled && !context.params.readOnly &&
786 detail::numeric::validStep(context.params.step) &&
787 context.params.value != nullptr &&
788 detail::numeric::isFinite(*context.params.value);
789 const FSELNumberInputStepButtonStateTheme appearance =
790 resolveStepAppearance(context, id, enabled, theme);
791 Clay_ElementDeclaration button{};
792 button.layout.sizing = {
793 .width = context.params.stepButtons ==
794 NumberInputStepButtons::TrailingHorizontal
795 ? CLAY_SIZING_FIXED(buttonWidth)
796 : CLAY_SIZING_GROW(0),
797 .height = CLAY_SIZING_GROW(0),
798 };
799 button.layout.childAlignment = {
800 .x = CLAY_ALIGN_X_CENTER,
801 .y = CLAY_ALIGN_Y_CENTER,
802 };
803 button.backgroundColor = appearance.backgroundColor;
804
805 TextureRef icon = increment
806 ? context.params.incrementIcon.value_or(
807 context.resources().incrementIcon)
808 : context.params.decrementIcon.value_or(
809 context.resources().decrementIcon);
810 CLAY(id, button) {
811 if (icon.handle) {
812 icon.tintEnabled = true;
813 const float iconSize = std::max(
814 context.params.stepIconSize.value_or(theme.stepIconSize),
815 1.0f);
816 Clay_ElementDeclaration iconDeclaration{};
817 iconDeclaration.layout.sizing = {
818 .width = CLAY_SIZING_FIXED(iconSize),
819 .height = CLAY_SIZING_FIXED(iconSize),

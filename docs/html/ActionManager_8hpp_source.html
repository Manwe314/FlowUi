511template <typename Operation, typename... Resources>
512UiActionCall UiActions::make(
513 UiActionRecipe<Operation> recipe,
514 Resources&&... resources) const noexcept {
515 (void)owner_;
516 using Traits = detail::ui_action::CallableSignature<decltype(&Operation::operator())>;
517 using Parameters = typename Traits::Arguments;
518 using Arguments = std::tuple<Resources&&...>;
519 constexpr std::size_t parameterCount = std::tuple_size_v<Parameters>;
520 static_assert(parameterCount == sizeof...(Resources),
521 "FlowUi UiAction resource count does not match the recipe parameter count.");
522
523 if constexpr (parameterCount == sizeof...(Resources)) {
524 constexpr auto indexes = std::make_index_sequence<parameterCount>{};
525 static_assert(
526 detail::ui_action::argumentsCanBind<Parameters, Arguments>(indexes),
527 "FlowUi UiAction resources do not satisfy the recipe parameter types; references require matching lvalues.");
528 static_assert([]<std::size_t... Index>(std::index_sequence<Index...>) {
529 return ((std::is_lvalue_reference_v<std::tuple_element_t<Index, Parameters>> ||
530 std::is_nothrow_constructible_v<
531 detail::ui_action::StoredArgumentT<std::tuple_element_t<Index, Parameters>>,
532 std::tuple_element_t<Index, Arguments>>) && ...);
533 }(indexes), "FlowUi UiAction argument conversions must not throw; convert fallible values before packing the action.");
534 static_assert(
535 detail::ui_action::StoredBytes<Parameters> <= UiActionCall::InlineBytes,
536 "FlowUi UiAction arguments exceed inline storage; borrow a context object or use an element-specific callback.");
537
538 UiActionCall call{};
539 auto arguments = std::forward_as_tuple(std::forward<Resources>(resources)...);
540 detail::ui_action::packArguments<Parameters>(
541 call.payload_.data(), std::move(arguments), indexes);
542 call.invokeThunk_ = &detail::ui_action::invokeRecipe<Operation>;
543#if FLOW_UI_DEV_MODE
544 call.recipeId_ = recipe.recipeId;
545 call.reconstructable_ = parameterCount == 0u;
546 call.recipeName_ = recipe.debugName;
547 call.definitionSource_ = recipe.definitionSource;
548#endif
549#if FLOW_UI_DEV_MODE || FLOW_UI_HAS_BAKED_CHANGES
550 owner_->noteUiRecipe(
551 recipe.recipeId, parameterCount == 0u ? &call : nullptr
552#if FLOW_UI_DEV_MODE
553 , recipe.debugName, recipe.definitionSource
554#endif
555 );
556#else
557 (void)recipe;
558#endif
559 return call;
560 } else {
561 return {};
562 }

604template <typename Callable, typename... Resources>
605Result<AppActionCall> ActionManager::bindTyped(
606 AppActionDescriptor descriptor,
607 bool replacement,
608 Callable&& callable,
609 Resources&&... resources) {
610 using StoredCallable = std::decay_t<Callable>;
611 static_assert(detail::action::StorableCallable<StoredCallable>,
612 "FlowUi app actions require a function/member pointer, stateless callable, or explicitly borrowed callable.");
613 static_assert((detail::action::BorrowedResource<std::decay_t<Resources>> && ...),
614 "FlowUi app action resources are borrowed; pass pointers or std::ref/std::cref.");
615 using Payload = detail::action::AppActionBindingPayload<
616 StoredCallable,
617 std::decay_t<Resources>...>;
618 using PayloadResult = detail::action::PayloadResult<Payload>;
619 static_assert(!std::is_reference_v<PayloadResult> && !std::is_array_v<PayloadResult>,
620 "FlowUi app action results cannot be references or arrays; return a pointer or reference_wrapper.");
621 static_assert(std::is_void_v<PayloadResult> || std::is_move_constructible_v<PayloadResult>,
622 "FlowUi app action results must be move constructible.");
623 static_assert(std::is_void_v<PayloadResult> || std::is_nothrow_destructible_v<PayloadResult>,
624 "FlowUi owned app action results must have non-throwing destructors.");
625
626 if (!storage_) return unexpectedError(makeError(ErrorCode::ObjectNotInitialized, ErrorSite::ActionBind));
627 auto validation = validateBindingRequest(descriptor.id, replacement);
628 if (!validation) return unexpectedError(validation.error());
629 const std::string_view effectiveName = descriptor.debugName.empty()
630 ? std::string_view{"flowui.app_action.binding"}
631 : descriptor.debugName;
632 const detail::storage::StringId debugName = storage_->intern(effectiveName);
633 detail::storage::PersistentRecordHandle handle{};
634 try {
635 handle = detail::storage::createTypedPersistentRecord<
636 detail::action::AppActionBindingHeader,
637 Payload>(
638 *storage_,
639 detail::storage::PersistentRecordCreateInfo{
640 .kind = detail::storage::ResourceKind::AppActionBinding,
641 .window = InvalidWindowId,
642 .debugName = debugName,
643 },
644 [descriptor, debugName](
645 detail::action::AppActionBindingHeader& header,
646 const detail::storage::AlignedRecordLayout&) {
647 header.id = descriptor.id;
648 header.callableTypeHash = detail::typeHash<StoredCallable>();
649 header.resultTypeHash = detail::action::resultTypeHash<Payload>();
650 header.debugName = debugName;
651 header.invokeDiscard = &detail::action::invokeDiscard<Payload>;
652 if constexpr (!std::is_void_v<PayloadResult>) {
653 header.invokeResult = &detail::action::invokeResult<Payload>;
654 }
655 },
656 StoredCallable(std::forward<Callable>(callable)),
657 std::decay_t<Resources>(std::forward<Resources>(resources))...);
658 } catch (const std::bad_alloc&) {
659 throw;
660 } catch (const FlowUiException& exception) {
661 if (exception.error().descriptor().category == ErrorCategory::Fatal) {
662 detail::terminateForFatalError(exception.error());
663 }
664 if (exception.error().descriptor().category != ErrorCategory::Local) throw;
665 return unexpectedError(exception.error());
666 } catch (...) {
667 return unexpectedError(makeError(ErrorCode::ResourceCreationFailed, ErrorSite::ActionBind));
668 }
669
670 try {
671 publishBinding(descriptor.id, handle, replacement);
672 } catch (const FlowUiException&) {
673 (void)storage_->removePersistentRecord(
674 handle, detail::storage::ResourceKind::AppActionBinding);
675 detail::terminateForFatalError(makeError(
676 ErrorCode::ActionPublicationConflict, ErrorSite::ActionPublish,
677 descriptor.id.value));
678 } catch (...) {
679 (void)storage_->removePersistentRecord(
680 handle, detail::storage::ResourceKind::AppActionBinding);
681 return unexpectedError(makeError(ErrorCode::ResourcePublicationFailed, ErrorSite::ActionPublish));
682 }
683 return AppActionCall{descriptor.id};
684}

686template <typename Result>
687ActionResult<Result> AppActions::invokeFor(AppActionCall call) {
688 static_assert(!std::is_reference_v<Result> && !std::is_array_v<Result>,
689 "FlowUi invokeFor results cannot be references or arrays.");
690 static_assert(std::is_void_v<Result> || std::is_nothrow_destructible_v<Result>,
691 "FlowUi invokeFor result cleanup must not throw.");
692 ActionInvokeError error = ActionInvokeError::Empty;
693 const uint64_t resultType = std::is_void_v<Result> ? 0 : detail::typeHash<Result>();
694 auto lease = owner_->beginInvocation(
695 call, resultType, true, ActionInvocationSource{}, error);
696 if (!lease) return ActionResult<Result>{error};
697
698 if constexpr (std::is_void_v<Result>) {
699 try {
700 lease.header().invokeDiscard(lease.payload());
701 lease.noteSuccess(false);
702 return ActionResult<void>::success();
703 } catch (...) {
704 lease.noteException();
705 throw;
706 }
707 } else {
708 alignas(Result) std::byte resultStorage[sizeof(Result)];
709 Result* result = nullptr;
710 try {
711 lease.header().invokeResult(lease.payload(), resultStorage);
712 result = reinterpret_cast<Result*>(resultStorage);
713 ActionResult<Result> output{std::move(*result)};
714 std::exchange(result, nullptr)->~Result();
715 lease.noteSuccess(false);
716 return output;
717 } catch (...) {
718 if (result) result->~Result();
719 lease.noteException();
720 throw;
721 }
722 }

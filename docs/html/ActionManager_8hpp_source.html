241template <typename Operation, typename... Resources>
242UiActionCall UiActions::make(
243 UiActionRecipe<Operation> recipe,
244 Resources&&... resources) const {
245 (void)owner_;
246 using Traits = detail::ui_action::CallableSignature<decltype(&Operation::operator())>;
247 using Parameters = typename Traits::Arguments;
248 using Arguments = std::tuple<Resources&&...>;
249 constexpr std::size_t parameterCount = std::tuple_size_v<Parameters>;
250 static_assert(parameterCount == sizeof...(Resources),
251 "FlowUi UiAction resource count does not match the recipe parameter count.");
252
253 if constexpr (parameterCount == sizeof...(Resources)) {
254 constexpr auto indexes = std::make_index_sequence<parameterCount>{};
255 static_assert(
256 detail::ui_action::argumentsCanBind<Parameters, Arguments>(indexes),
257 "FlowUi UiAction resources do not satisfy the recipe parameter types; references require matching lvalues.");
258 static_assert(
259 detail::ui_action::StoredBytes<Parameters> <= UiActionCall::InlineBytes,
260 "FlowUi UiAction arguments exceed inline storage; borrow a context object or use an element-specific callback.");
261
262 UiActionCall call{};
263 auto arguments = std::forward_as_tuple(std::forward<Resources>(resources)...);
264 detail::ui_action::packArguments<Parameters>(
265 call.payload_.data(), std::move(arguments), indexes);
266 call.invokeThunk_ = &detail::ui_action::invokeRecipe<Operation>;
267#if FLOW_UI_DEV_MODE
268 call.recipeId_ = recipe.recipeId;
269 call.reconstructable_ = parameterCount == 0u;
270 call.recipeName_ = recipe.debugName;
271 call.definitionSource_ = recipe.definitionSource;
272#endif
273#if FLOW_UI_DEV_MODE || FLOW_UI_HAS_BAKED_CHANGES
274 owner_->noteUiRecipe(
275 recipe.recipeId, parameterCount == 0u ? &call : nullptr
276#if FLOW_UI_DEV_MODE
277 , recipe.debugName, recipe.definitionSource
278#endif
279 );
280#else
281 (void)recipe;
282#endif
283 return call;
284 } else {
285 return {};
286 }

328template <typename Callable, typename... Resources>
329Result<AppActionCall> ActionManager::bindTyped(
330 AppActionDescriptor descriptor,
331 bool replacement,
332 Callable&& callable,
333 Resources&&... resources) {
334 using StoredCallable = std::decay_t<Callable>;
335 static_assert(detail::action::StorableCallable<StoredCallable>,
336 "FlowUi app actions require a function/member pointer, stateless callable, or explicitly borrowed callable.");
337 static_assert((detail::action::BorrowedResource<std::decay_t<Resources>> && ...),
338 "FlowUi app action resources are borrowed; pass pointers or std::ref/std::cref.");
339 using Payload = detail::action::AppActionBindingPayload<
340 StoredCallable,
341 std::decay_t<Resources>...>;
342 using PayloadResult = detail::action::PayloadResult<Payload>;
343 static_assert(!std::is_reference_v<PayloadResult> && !std::is_array_v<PayloadResult>,
344 "FlowUi app action results cannot be references or arrays; return a pointer or reference_wrapper.");
345 static_assert(std::is_void_v<PayloadResult> || std::is_move_constructible_v<PayloadResult>,
346 "FlowUi app action results must be move constructible.");
347
348 if (!storage_) return unexpectedError(makeError(ErrorCode::ObjectNotInitialized, ErrorSite::ActionBind));
349 auto validation = validateBindingRequest(descriptor.id, replacement);
350 if (!validation) return unexpectedError(validation.error());
351 const std::string_view effectiveName = descriptor.debugName.empty()
352 ? std::string_view{"flowui.app_action.binding"}
353 : descriptor.debugName;
354 const detail::storage::StringId debugName = storage_->intern(effectiveName);
355 detail::storage::PersistentRecordHandle handle{};
356 try {
357 handle = detail::storage::createTypedPersistentRecord<
358 detail::action::AppActionBindingHeader,
359 Payload>(
360 *storage_,
361 detail::storage::PersistentRecordCreateInfo{
362 .kind = detail::storage::ResourceKind::AppActionBinding,
363 .window = InvalidWindowId,
364 .debugName = debugName,
365 },
366 [descriptor, debugName](
367 detail::action::AppActionBindingHeader& header,
368 const detail::storage::AlignedRecordLayout&) {
369 header.id = descriptor.id;
370 header.callableTypeHash = detail::typeHash<StoredCallable>();
371 header.resultTypeHash = detail::action::resultTypeHash<Payload>();
372 header.debugName = debugName;
373 header.invokeDiscard = &detail::action::invokeDiscard<Payload>;
374 if constexpr (!std::is_void_v<PayloadResult>) {
375 header.invokeResult = &detail::action::invokeResult<Payload>;
376 }
377 },
378 StoredCallable(std::forward<Callable>(callable)),
379 std::decay_t<Resources>(std::forward<Resources>(resources))...);
380 } catch (const std::bad_alloc&) {
381 throw;
382 } catch (const FlowUiException& exception) {
383 if (exception.error().descriptor().category == ErrorCategory::Fatal) {
384 detail::terminateForFatalError(exception.error());
385 }
386 if (exception.error().descriptor().category != ErrorCategory::Local) throw;
387 return unexpectedError(exception.error());
388 } catch (...) {
389 return unexpectedError(makeError(ErrorCode::ResourceCreationFailed, ErrorSite::ActionBind));
390 }
391
392 try {
393 publishBinding(descriptor.id, handle, replacement);
394 } catch (const FlowUiException&) {
395 (void)storage_->removePersistentRecord(
396 handle, detail::storage::ResourceKind::AppActionBinding);
397 detail::terminateForFatalError(makeError(
398 ErrorCode::ActionPublicationConflict, ErrorSite::ActionPublish,
399 descriptor.id.value));
400 } catch (...) {
401 (void)storage_->removePersistentRecord(
402 handle, detail::storage::ResourceKind::AppActionBinding);
403 return unexpectedError(makeError(ErrorCode::ResourcePublicationFailed, ErrorSite::ActionPublish));
404 }
405 return AppActionCall{descriptor.id};
406}

408template <typename Result>
409ActionResult<Result> AppActions::invokeFor(AppActionCall call) {
410 static_assert(!std::is_reference_v<Result> && !std::is_array_v<Result>,
411 "FlowUi invokeFor results cannot be references or arrays.");
412 ActionInvokeError error = ActionInvokeError::Empty;
413 const uint64_t resultType = std::is_void_v<Result> ? 0 : detail::typeHash<Result>();
414 auto lease = owner_->beginInvocation(
415 call, resultType, true, ActionInvocationSource{}, error);
416 if (!lease) return ActionResult<Result>{error};
417
418 if constexpr (std::is_void_v<Result>) {
419 try {
420 lease.header().invokeDiscard(lease.payload());
421 lease.noteSuccess(false);
422 return ActionResult<void>::success();
423 } catch (...) {
424 lease.noteException();
425 throw;
426 }
427 } else {
428 alignas(Result) std::byte resultStorage[sizeof(Result)];
429 Result* result = nullptr;
430 try {
431 lease.header().invokeResult(lease.payload(), resultStorage);
432 result = reinterpret_cast<Result*>(resultStorage);
433 ActionResult<Result> output{std::move(*result)};
434 result->~Result();
435 lease.noteSuccess(false);
436 return output;
437 } catch (...) {
438 if (result) result->~Result();
439 lease.noteException();
440 throw;
441 }
442 }

155template<NumericValueType T>
156[[nodiscard]] T stepValue(
157 T value,
158 T step,
159 int direction,
160 Bounds<T> bounds) noexcept {
161 if (!bounds.valid) {
162 return value;
163 }
164 if (!validStep(step) || direction == 0) {
165 return clampValue(value, bounds);
166 }
167
168 value = clampValue(value, bounds);
169 if (direction > 0) {
170 if (value >= bounds.upper) {
171 return bounds.upper;
172 }
173 if constexpr (std::integral<T>) {
174 const T nativeMaximum = std::numeric_limits<T>::max();
175 const T candidate = value > nativeMaximum - step
176 ? nativeMaximum
177 : static_cast<T>(value + step);
178 return std::min(candidate, bounds.upper);
179 } else {
180 const T candidate = static_cast<T>(value + step);
181 return !isFinite(candidate)
182 ? bounds.upper
183 : std::min(candidate, bounds.upper);
184 }
185 }
186
187 if (value <= bounds.lower) {
188 return bounds.lower;
189 }
190 if constexpr (std::integral<T>) {
191 if constexpr (std::unsigned_integral<T>) {
192 const T candidate = value < step ? T{} : static_cast<T>(value - step);
193 return std::max(candidate, bounds.lower);
194 } else {
195 const T nativeMinimum = std::numeric_limits<T>::lowest();
196 const T candidate = value < nativeMinimum + step
197 ? nativeMinimum
198 : static_cast<T>(value - step);
199 return std::max(candidate, bounds.lower);
200 }
201 } else {
202 const T candidate = static_cast<T>(value - step);
203 return !isFinite(candidate)
204 ? bounds.lower
205 : std::max(candidate, bounds.lower);
206 }
207}

209/* Apply a signed number of native steps without overflowing intermediate values. */
210template<NumericValueType T>
211[[nodiscard]] T offsetBySteps(
212 T value,
213 T step,
214 int64_t stepCount,
215 Bounds<T> bounds) noexcept {
216 if (!bounds.valid || !validStep(step) || stepCount == 0) {
217 return bounds.valid ? clampValue(value, bounds) : value;
218 }
219
220 const bool increasing = stepCount > 0;
221 const uint64_t magnitude = stepCount < 0
222 ? static_cast<uint64_t>(-(stepCount + 1)) + 1u
223 : static_cast<uint64_t>(stepCount);
224 value = clampValue(value, bounds);
225
226 if constexpr (std::floating_point<T>) {
227 const long double signedMagnitude = increasing
228 ? static_cast<long double>(magnitude)
229 : -static_cast<long double>(magnitude);
230 const long double candidate = static_cast<long double>(value) +
231 static_cast<long double>(step) * signedMagnitude;
232 if (!std::isfinite(candidate)) {
233 return increasing ? bounds.upper : bounds.lower;
234 }
235 return static_cast<T>(std::clamp(
236 candidate,
237 static_cast<long double>(bounds.lower),
238 static_cast<long double>(bounds.upper)));
239 } else {
240 using Unsigned = std::make_unsigned_t<T>;
241 static_assert(sizeof(Unsigned) <= sizeof(uint64_t));
242 const Unsigned unsignedStep = static_cast<Unsigned>(step);
243 const Unsigned distance = increasing
244 ? static_cast<Unsigned>(bounds.upper) - static_cast<Unsigned>(value)
245 : static_cast<Unsigned>(value) - static_cast<Unsigned>(bounds.lower);
246 if (distance == 0u) {
247 return increasing ? bounds.upper : bounds.lower;
248 }
249 const Unsigned requiredSteps = static_cast<Unsigned>(
250 distance / unsignedStep +
251 (distance % unsignedStep == 0u ? 0u : 1u));
252 if (magnitude >= static_cast<uint64_t>(requiredSteps)) {
253 return increasing ? bounds.upper : bounds.lower;
254 }
255 const Unsigned offset = static_cast<Unsigned>(
256 unsignedStep * static_cast<Unsigned>(magnitude));
257
258 if constexpr (std::unsigned_integral<T>) {
259 return increasing
260 ? static_cast<T>(value + offset)
261 : static_cast<T>(value - offset);
262 } else {
263 auto fromNegativeMagnitude = [](Unsigned negativeMagnitude) {
264 if (negativeMagnitude == 0u) {
265 return T{};
266 }
267 return static_cast<T>(
268 -static_cast<T>(negativeMagnitude - 1u) - 1);
269 };
270
271 if (increasing) {
272 if (value >= T{}) {
273 return static_cast<T>(
274 value + static_cast<T>(offset));
275 }
276 const Unsigned negativeMagnitude =
277 static_cast<Unsigned>(-(value + 1)) + 1u;
278 if (offset < negativeMagnitude) {
279 return fromNegativeMagnitude(
280 negativeMagnitude - offset);
281 }
282 return static_cast<T>(offset - negativeMagnitude);
283 }
284
285 if (value <= T{}) {
286 const Unsigned negativeMagnitude =
287 static_cast<Unsigned>(-(value + 1)) + 1u;
288 return fromNegativeMagnitude(negativeMagnitude + offset);
289 }
290 const Unsigned positiveValue = static_cast<Unsigned>(value);
291 if (offset <= positiveValue) {
292 return static_cast<T>(positiveValue - offset);
293 }
294 return fromNegativeMagnitude(offset - positiveValue);
295 }
296 }

151 static void onHovered(InteractionContext& context) {
152 if (!isEnabled(context)) {
153 return;
154 }
155
156 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().buttonTheme;
157 context.uiManager.requestCursor(
158 context.params.cursor.value_or(theme.cursor),
159 context.params.cursorPriority.value_or(theme.cursorPriority));
160 }

166 static void onPressed(InteractionContext& context) {
167 auto& state = context.state();
168 if (!isEnabled(context)) {
169 clearInteraction(state);
170 return;
171 }
172
173 state.isArmed = true;
174 state.mouseUpObserved = false;
175 }
176
181 static void onReleased(InteractionContext& context) {
182 auto& state = context.state();
183 if (!state.isArmed) {
184 return;
185 }
186
187 const bool shouldActivate = isEnabled(context);
188 clearInteraction(state);
189 if (shouldActivate) {
190 (void)context.invoke(context.params.onActivate);
191 }
192 }
193
198 static void runLogic(InteractionContext& context) {
199 auto& state = context.state();
200 if (!state.isArmed) {
201 return;
202 }
203 if (!isEnabled(context)) {
204 clearInteraction(state);
205 return;
206 }
207
208 const bool pointerDown =
209 context.uiManager.getCurrentFrameInput().mouseDown[0];
210 if (pointerDown) {
211 if (state.mouseUpObserved) {
212 clearInteraction(state);
213 }
214 return;
215 }
216
217 if (state.mouseUpObserved) {
218 clearInteraction(state);
219 } else {
220 state.mouseUpObserved = true;
221 }
222 }

229 static Clay_ElementDeclaration constructElement(BuildContext& context) {
230 const FSELButtonStateTheme appearance = resolveAppearance(context);
231 return makeRootDeclaration(context, appearance);
232 }
233
238 static void buildElement(BuildContext& context) {
239 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().buttonTheme;
240 const FSELButtonStateTheme appearance = resolveAppearance(context);
241 const Clay_ElementDeclaration root = makeRootDeclaration(context, appearance);
242 const bool needsText =
243 context.params.contentMode == ButtonContentMode::TextOnly ||
244 context.params.contentMode == ButtonContentMode::IconThenText ||
245 context.params.contentMode == ButtonContentMode::TextThenIcon;
246 const bool needsIcon =
247 context.params.contentMode == ButtonContentMode::IconOnly ||
248 context.params.contentMode == ButtonContentMode::IconThenText ||
249 context.params.contentMode == ButtonContentMode::TextThenIcon;
250
251 Clay_TextElementConfig labelConfig{};
252 Clay_ElementDeclaration iconDeclaration{};
253 if (needsText) {
254 labelConfig.textColor = appearance.labelColor;
255 labelConfig.fontId = context.uiManager.resolveFont(
256 context.params.labelFontFamily.value_or(theme.labelFontFamily),
257 context.params.labelFontWeight.value_or(theme.labelFontWeight),
258 context.params.labelFontStyle.value_or(theme.labelFontStyle));
259 labelConfig.fontSize = context.params.labelFontSize.value_or(
260 theme.labelFontSize);
261 labelConfig.wrapMode = context.params.labelWrapMode.value_or(
262 theme.labelWrapMode);
263 labelConfig.textAlignment = context.params.labelAlignment.value_or(
264 theme.labelAlignment);
265 }
266 if (needsIcon) {
267 TextureRef icon = context.params.icon;
268 icon.tintEnabled = context.params.tintIcon;
269 const float iconSize = std::max(
270 context.params.iconSize.value_or(theme.iconSize),
271 1.0f);
272 iconDeclaration.layout.sizing = {
273 .width = CLAY_SIZING_FIXED(iconSize),
274 .height = CLAY_SIZING_FIXED(iconSize),
275 };
276 iconDeclaration.backgroundColor = appearance.iconColor;
277 iconDeclaration.image = {
278 .imageData = context.uiManager.imageData(icon),
279 };
280 }
281
282 auto drawText = [&]() {
283 CLAY(FlowUi::ValueOrThrow(context.clayID("label")), {}) {
284 CLAY_TEXT(
285 context.uiManager.toClayString(context.params.text),
286 CLAY_TEXT_CONFIG(labelConfig));
287 }
288 };
289 auto drawIcon = [&]() {
290 CLAY(FlowUi::ValueOrThrow(context.clayID("icon")), iconDeclaration);
291 };
292
293 CLAY(context.clayID(), root) {
294 switch (context.params.contentMode) {
295 case ButtonContentMode::None:
296 break;
297 case ButtonContentMode::TextOnly:
298 drawText();
299 break;
300 case ButtonContentMode::IconOnly:
301 drawIcon();
302 break;
303 case ButtonContentMode::IconThenText:
304 drawIcon();
305 drawText();
306 break;
307 case ButtonContentMode::TextThenIcon:
308 drawText();
309 drawIcon();
310 break;
311 }
312 }
313 }

339 static VisualState resolveVisualState(BuildContext& context) {
340 if (!isEnabled(context)) {
341 return VisualState::Disabled;
342 }
343
344 const bool hovered = context.uiManager
345 .getPreviousFramesInteraction()
346 .isHovered(context.clayID());
347 const bool pressed = context.state().isArmed && hovered &&
348 context.uiManager.getCurrentFrameInput().mouseDown[0];
349 if (pressed) {
350 return VisualState::Pressed;
351 }
352 return hovered ? VisualState::Hovered : VisualState::Idle;
353 }

355 static FSELButtonStateTheme applyOverrides(

361 appearance.labelColor);

369 static FSELButtonStateTheme resolveAppearance(BuildContext& context) {
370 const auto& theme = FlowUi::ValueOrThrow(context.uiManager.theme<FSELTheme>()).get().buttonTheme;
371 switch (resolveVisualState(context)) {
372 case VisualState::Idle:
373 return applyOverrides(theme.idle, context.params.idleOverrides);
374 case VisualState::Hovered:
375 return applyOverrides(theme.hovered, context.params.hoveredOverrides);
376 case VisualState::Pressed:
377 return applyOverrides(theme.pressed, context.params.pressedOverrides);
378 case VisualState::Disabled:
379 return applyOverrides(theme.disabled, context.params.disabledOverrides);
380 }
381 return applyOverrides(theme.idle, context.params.idleOverrides);
382 }

400 .color = appearance.borderColor,
401 .width = context.params.borderWidth.value_or(theme.borderWidth),
402 };

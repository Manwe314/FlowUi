273[[nodiscard]] constexpr auto UiAction(
274 Fn fn
275#if FLOW_UI_DEV_MODE
276 , ActionSourceLocation source = ActionSourceLocation::current()
277#endif
278 ) {
279 using Operation = std::remove_cvref_t<Fn>;
280 static_assert(std::is_empty_v<Operation>,
281 "FlowUi UiAction recipes must be stateless; pass local resources to uiActions().make().");
282 static_assert(std::is_trivially_copyable_v<Operation> &&
283 std::is_trivially_destructible_v<Operation> &&
284 std::is_default_constructible_v<Operation>,
285 "FlowUi UiAction recipes must be trivial and default constructible.");
286 static_assert(detail::ui_action::ConcreteRecipeCallable<Operation>,
287 "FlowUi UiAction recipes need one concrete, non-generic call operator.");
288 using Traits = detail::ui_action::CallableSignature<decltype(&Operation::operator())>;
289 static_assert(std::is_void_v<typename Traits::Result>,
290 "FlowUi UiAction recipes must return void.");
291 return UiActionRecipe<Operation>{
292 .operation = fn,
293 .recipeId = detail::ui_action::callableTypeId<Operation>(),
294#if FLOW_UI_DEV_MODE
295 .debugName = {},
296 .definitionSource = source,
297#endif
298 };
299}

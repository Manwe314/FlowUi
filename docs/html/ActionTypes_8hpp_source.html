450template <typename Fn>
451[[nodiscard]] constexpr auto UiAction(
452 Fn fn
453#if FLOW_UI_DEV_MODE
454 , ActionSourceLocation source = ActionSourceLocation::current()
455#endif
456 ) noexcept {
457 using Operation = std::remove_cvref_t<Fn>;
458 static_assert(std::is_empty_v<Operation>,
459 "FlowUi UiAction recipes must be stateless; pass local resources to uiActions().make().");
460 static_assert(std::is_trivially_copyable_v<Operation> &&
461 std::is_trivially_destructible_v<Operation> &&
462 std::is_default_constructible_v<Operation>,
463 "FlowUi UiAction recipes must be trivial and default constructible.");
464 static_assert(detail::ui_action::ConcreteRecipeCallable<Operation>,
465 "FlowUi UiAction recipes need one concrete, non-generic call operator.");
466 using Traits = detail::ui_action::CallableSignature<decltype(&Operation::operator())>;
467 static_assert(std::is_void_v<typename Traits::Result>,
468 "FlowUi UiAction recipes must return void.");
469 return UiActionRecipe<Operation>{
470 .operation = fn,
471 .recipeId = detail::ui_action::callableTypeId<Operation>(),
472#if FLOW_UI_DEV_MODE
473 .debugName = {},
474 .definitionSource = source,
475#endif
476 };
477}

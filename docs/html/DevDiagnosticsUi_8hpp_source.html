60template <class Context, class Identity>
61void pointer_tooltip(Context& context, const Identity& identity, std::string_view description) {
62 const auto& input = context.uiManager.getCurrentFrameInput();
63 if (input.mouseX < 0 || input.mouseY < 0)
64 return;
65 const auto dimensions = Clay_GetLayoutDimensions();
66 const auto previous = Clay_GetElementData(context.clayID(identity));
67 const float width = std::min(300.0f, std::max(80.0f, dimensions.width - 16));
68 const float height = previous.found ? previous.boundingBox.height : 160;
69 constexpr float gap = 12;
70 const bool above = input.mouseY >= height + gap + 8;
71 auto tooltip = card();
72 tooltip.layout.sizing.width = CLAY_SIZING_FIXED(width);
73 tooltip.floating.attachTo = CLAY_ATTACH_TO_ROOT;
74 tooltip.floating.attachPoints = {above ? CLAY_ATTACH_POINT_LEFT_BOTTOM
75 : CLAY_ATTACH_POINT_LEFT_TOP,
76 CLAY_ATTACH_POINT_LEFT_TOP};
77 tooltip.floating.offset = {
78 std::clamp(input.mouseX + gap, 8.0f, std::max(8.0f, dimensions.width - width - 8)),
79 above ? input.mouseY - gap : input.mouseY + gap};
80 tooltip.floating.zIndex = 140;
81 tooltip.floating.clipTo = CLAY_CLIP_TO_NONE;
82 tooltip.floating.pointerCaptureMode = CLAY_POINTER_CAPTURE_MODE_PASSTHROUGH;
83 CLAY(context.clayID(identity), tooltip)
84 text(context, description, interface_theme::kTextCanvas, 11);
85}

250 static void buildElement(BuildContext& context) {
251 const auto& parameters = context.params;
252 const auto surface = constructElement(context);
253 CLAY(context.clayID(), surface) {

255 auto headline = row(16);
256 if (parameters.flexible)

258 CLAY(context.clayID("headline"), headline) {
259 icon(context, "marker",
260 parameters.icon == diagnostics_icons::Icon::None
261 ? diagnostics_icons::Icon::Issue
262 : parameters.icon,
263 parameters.marker, 14);

267 label.clip.horizontal = true;
268 label.clip.scrollInputDisabled = true;
269 CLAY(context.clayID("label"), label) {
270 Clay_TextElementConfig configuration{};

272 configuration.textColor = interface_theme::kTextCanvas;
273 configuration.wrapMode =
274 parameters.flexible ? CLAY_TEXT_WRAP_WORDS : CLAY_TEXT_WRAP_NONE;
275 CLAY_TEXT(context.uiManager.toClayString(parameters.label),
276 CLAY_TEXT_CONFIG(configuration));
277 }
278 text(context, parameters.badge, interface_theme::kTextSecondary, 11);
279 }
280 if (parameters.flexible) {
281 text(context, parameters.secondary, interface_theme::kTextSecondary, 12);
282 } else {
283 auto summary = column(0);
284 summary.layout.sizing.height = CLAY_SIZING_FIXED(14);
285 summary.clip.horizontal = true;
286 summary.clip.scrollInputDisabled = true;
287 CLAY(context.clayID("summary"), summary) {
288 Clay_TextElementConfig configuration{};
289 configuration.fontSize = 11;
290 configuration.textColor = interface_theme::kTextSecondary;
291 configuration.wrapMode = CLAY_TEXT_WRAP_NONE;
292 CLAY_TEXT(context.uiManager.toClayString(parameters.secondary),
293 CLAY_TEXT_CONFIG(configuration));
294 }
295 }
296 } else {
297 const auto semantic = !parameters.clipboard_text.empty() &&
298 parameters.icon == diagnostics_icons::Icon::None
299 ? diagnostics_icons::Icon::Copy
300 : parameters.icon;
301 bool has_icon = false;
302#if FLOWUI_INCLUDE_ICON_MANAGER
303 has_icon = static_cast<bool>(context.resources().texture(semantic).handle);
304#endif
305 const bool icon_only = parameters.icon_only || !parameters.clipboard_text.empty();
306 if (!icon_only || has_icon)
307 icon(context, "action-icon", semantic,
308 parameters.enabled ? interface_theme::kTextSecondary
309 : interface_theme::kTextMuted,
310 parameters.section_header ? 12 : 14);
311 if (!icon_only || !has_icon)
312 text(context, parameters.label,
313 parameters.enabled ? interface_theme::kTextCanvas
314 : interface_theme::kTextMuted,
315 12);
316 }
317 const std::string_view tooltip =
318 parameters.tooltip.empty() &&
319 (parameters.icon_only || !parameters.clipboard_text.empty())
320 ? parameters.label
321 : parameters.tooltip;
322 if (!tooltip.empty() &&
323 context.uiManager.getPreviousFramesInteraction().isHovered(context.clayID())) {
324 pointer_tooltip(context, LocalElementName{"tooltip"}, tooltip);
325 }
326 }

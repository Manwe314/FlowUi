129[[nodiscard]] constexpr const DevMemoryNodeDescriptor* memory_node(uint64_t identity) noexcept {
130 for (const auto& node : memory_selector_nodes)
131 if (static_cast<uint64_t>(node.id) == identity)
132 return &node;
133 return nullptr;
134}
136[[nodiscard]] constexpr bool memory_node_has_children(DevMemoryNodeId identity) noexcept {
137 if (identity == DevMemoryNodeId::GpuHeaps)
138 return true;
139 for (const auto& node : memory_selector_nodes)
140 if (node.parent == identity)
141 return true;
142 return false;
143}
145[[nodiscard]] constexpr bool memory_node_descends_from(uint64_t identity,
146 DevMemoryNodeId ancestor) noexcept {
147 if (identity >= memory_heap_node_base)
148 return ancestor == DevMemoryNodeId::GpuHeaps;
149 const auto* node = memory_node(identity);
150 while (node && node->parent != DevMemoryNodeId::None) {
151 if (node->parent == ancestor)
152 return true;
153 node = memory_node(static_cast<uint64_t>(node->parent));
154 }
155 return false;
156}
158[[nodiscard]] constexpr bool memory_selection_application_wide(uint64_t identity) noexcept {
159 const auto* node = memory_node(identity);
160 return identity >= memory_heap_node_base || (node && node->application_wide);
161}

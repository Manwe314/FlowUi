113struct Resources {
114 std::array<TextureRef, static_cast<std::size_t>(Icon::Count)> textures{};
115 explicit Resources([[maybe_unused]] App& app) {
116#if FLOWUI_INCLUDE_ICON_MANAGER
117 auto& icons = app.icons();
118 interface_icons::registerDevInterfaceIcons(icons);
119 const auto assign = [&](Icon icon, std::string_view key) {
120 textures[static_cast<std::size_t>(icon)] = icons.textureRef(key);
121 };
122 assign(Icon::Issue, interface_icons::kErrorReporterKey);
123 assign(Icon::Warning, interface_icons::kErrorReporterKey);
124 assign(Icon::Expanded, interface_icons::kTreeExpandedKey);
125 assign(Icon::Collapsed, interface_icons::kTreeCollapsedKey);
126 assign(Icon::Copy, interface_icons::kCopyKey);
127 assign(Icon::Refresh, interface_icons::kRefreshKey);
128 assign(Icon::Back, interface_icons::kBackKey);
129 assign(Icon::Undo, interface_icons::kUndoKey);
130 assign(Icon::Redo, interface_icons::kRedoKey);
131 assign(Icon::Remove, interface_icons::kRemoveKey);
132 assign(Icon::ZoomIn, interface_icons::kZoomInKey);
133 assign(Icon::ZoomOut, interface_icons::kZoomOutKey);
134 assign(Icon::Fit, interface_icons::kFitKey);
135 assign(Icon::Native, interface_icons::kPreviewResetKey);
136 for (const auto& descriptor : catalogue) {
137 if (!icons.contains(descriptor.key)) {
138 const auto registered = icons.registerSvg(descriptor.key, descriptor.svg);
139 if (!registered)
140 continue;
141 }
142 assign(descriptor.icon, descriptor.key);
143 }
144 textures[static_cast<std::size_t>(Icon::Snapshot)] =
145 textures[static_cast<std::size_t>(Icon::Capture)];
146#else
147 textures.fill(TextureRef{});
148#endif
149 }
151 [[nodiscard]] TextureRef texture(Icon icon) const noexcept {
152 const auto index = static_cast<std::size_t>(icon);
153 return index < textures.size() ? textures[index] : TextureRef{};
154 }
155};

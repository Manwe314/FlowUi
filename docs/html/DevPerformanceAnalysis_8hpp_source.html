83[[nodiscard]] inline bool performance_sample_valid(const TimelineBlockSlice& block) noexcept {
84 if (block.synthetic_tick)
85 return false;
86 const auto flags = block.details().quality_flags;
87 if (block.domain == TimingSampleDomain::Gpu)
88 return (flags & gpuTimingRecordFlags(GpuTimingRecordFlag::Completed)) != 0;
89 constexpr auto rejected = timingRecordFlags(TimingRecordFlag::Canceled) |
90 timingRecordFlags(TimingRecordFlag::Incomplete) |
91 timingRecordFlags(TimingRecordFlag::ClockAnomaly) |
92 timingRecordFlags(TimingRecordFlag::OutOfDate) |
93 timingRecordFlags(TimingRecordFlag::Exception);
94 return (flags & timingRecordFlags(TimingRecordFlag::Completed)) && !(flags & rejected);
95}
98[[nodiscard]] inline PerformanceAnalysis analyze_performance(const TimingCaptureReadView& capture,
99 uint64_t tick_limit,
100 WindowId cadence_window) {
101 PerformanceAnalysis result;
102 result.generation = capture.generation;
103 std::vector<const TimingAppTickReport*> ticks;
104 ticks.reserve(capture.first.size() + capture.second.size());
105 for (const auto segment : {capture.first, capture.second})
106 for (const auto& tick : segment)
107 if (tick.occupied)
108 ticks.emplace_back(&tick);
109 const size_t first_position =
110 tick_limit && ticks.size() > tick_limit ? ticks.size() - tick_limit : 0;
111 if (first_position == ticks.size())
112 return result;
113 result.first_tick = ticks[first_position]->appTick;
114 result.last_tick = ticks.back()->appTick;
115 result.cadence.reserve(ticks.size() - first_position);
116 for (size_t tick_position = first_position; tick_position < ticks.size(); ++tick_position) {
117 const auto& tick = *ticks[tick_position];
118 const bool complete = !tick.boundary_open && tick.boundary_end_ns > tick.boundary_start_ns;
119 if (complete) {
120 ++result.complete_ticks;
121 result.coverage_ns += tick.boundary_end_ns - tick.boundary_start_ns;
122 if (!cadence_window)
123 result.cadence.emplace_back(tick.boundary_end_ns - tick.boundary_start_ns);
124 } else
125 ++result.excluded_intervals;
126 result.dropped_records += tick.cpuQuality.droppedRecords + tick.gpuQuality.droppedRecords;
127 if (cadence_window)
128 for (const auto& window : tick.windows)
129 if (window.occupied && window.window == cadence_window)
130 for (const auto& frame : window.frames)
131 if (frame.occupied)
132 for (const auto& sample : frame.cpuZones)
133 if (sample.typeId == timing_zones::kWindowFrameTotal.typeId &&
134 (sample.flags &
135 timingRecordFlags(TimingRecordFlag::Completed)) &&
136 !(sample.flags &
137 (timingRecordFlags(TimingRecordFlag::Incomplete) |
138 timingRecordFlags(TimingRecordFlag::ClockAnomaly) |
139 timingRecordFlags(TimingRecordFlag::Canceled))))
140 result.cadence.emplace_back(sample.durationNs);
141 }
142 result.sorted_cadence = result.cadence;
143 std::ranges::sort(result.sorted_cadence);
144 DevPerformanceSelection all_application;
145 const auto snapshot = extract_timeline(capture, all_application);
146 result.samples.reserve(snapshot.blocks.size());
147 for (const auto& block : snapshot.blocks) {
148 if (block.synthetic_tick || block.app_tick < result.first_tick ||
149 block.app_tick > result.last_tick)
150 continue;
151 if (!performance_sample_valid(block)) {
152 ++result.excluded_samples;
153 continue;
154 }
155 const auto details = block.details();
156 const bool self_valid =
157 block.domain == TimingSampleDomain::Cpu && block.hierarchy_note.empty() &&
158 !(details.quality_flags & timingRecordFlags(TimingRecordFlag::DetailTruncated)) &&
159 !block.cpu_sample.empty() &&
160 block.cpu_sample.front().directChildNs <= block.duration_ns;
161 result.samples.emplace_back(PerformanceAnalysisSample{
162 block.duration_ns, details.exclusive_ns, block.type_id, block.invocation_id,
163 block.app_tick, block.track, block.device_identity, block.queue_identity,
164 block.submission_serial, block.frame, block.zone_index, block.domain, self_valid});
165 }
166 std::ranges::sort(result.samples, [](const auto& left, const auto& right) {
167 return std::tuple{left.population(), left.app_tick, left.invocation_id, left.submission,
168 left.zone_index} < std::tuple{right.population(), right.app_tick,
169 right.invocation_id, right.submission,
170 right.zone_index};
171 });
172 result.zones.reserve(std::min(result.samples.size(), capture.descriptors.size() * 2));
173 for (size_t sample_position = 0; sample_position < result.samples.size();) {
174 const auto& first = result.samples[sample_position];
175 PerformanceAnalysisZone zone;
176 zone.first_sample = sample_position;
177 zone.largest = first;
178 const auto descriptor =
179 std::ranges::find(capture.descriptors, first.type_id, &TimingZoneDescriptor::typeId);
180 if (descriptor != capture.descriptors.end()) {
181 zone.label = performance_zone_label(descriptor->name);
182 zone.category = descriptor->category;
183 zone.role = descriptor->role;
184 } else
185 zone.label = "Unknown zone";
186 size_t population_end = sample_position;
187 while (population_end < result.samples.size() &&
188 result.samples[population_end].population() == first.population())
189 ++population_end;
190 zone.durations.reserve(population_end - sample_position);
191 for (; sample_position < population_end; ++sample_position) {
192 const auto& sample = result.samples[sample_position];
193 zone.durations.emplace_back(sample.duration_ns);
194 zone.total_ns += sample.duration_ns;
195 if (sample.self_valid) {
196 zone.self_ns += sample.exclusive_ns;
197 ++zone.self_count;
198 }
199 if (sample.duration_ns > zone.largest.duration_ns)
200 zone.largest = sample;
201 }
202 zone.sample_end = sample_position;
203 std::ranges::sort(zone.durations);
204 result.zones.emplace_back(std::move(zone));
205 }
206 return result;
207}
